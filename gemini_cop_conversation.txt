/**
 * packages/cop-kernel/src/storage.js
 * @file This module provides a robust and flexible storage interface for the COP kernel,
 * abstracting away the underlying database (Supabase) and offering caching, synchronization modes,
 * and debugging utilities. It follows a Singleton-like pattern to ensure a single, configured
 * storage instance is used throughout the application, while also acting as a Factory and Registry
 * for different storage configurations.
 * @module storage
 */

/**
 * @typedef {object} StorageInterface
 * @property {object} debugLogs - Interface for interacting with debug logs.
 * @property {function(object): Promise<object>} debugLogs.insert - Inserts a new debug log entry.
 * @property {object} events - Interface for interacting with events.
 * @property {function(object): Promise<object>} events.insert - Inserts a new event entry.
 * @property {object} artifacts - Interface for interacting with artifacts.
 * @property {function(object): Promise<object>} artifacts.insert - Inserts a new artifact entry.
 * @property {object} agentIdentities - Interface for interacting with agent identities.
 * @property {function(object): Promise<object>} agentIdentities.upsert - Inserts or updates an agent identity.
 * @property {function(string): Promise<object>} agentIdentities.getById - Retrieves an agent identity by ID.
 * @property {function(string): Promise<object>} agentIdentities.getByName - Retrieves an agent identity by name.
 * @property {function(object): Promise<object>} agentIdentities.list - Lists agent identities with optional filters.
 * @property {function(string, string): Promise<object>} agentIdentities.updateStatus - Updates the status of an agent identity.
 * @property {object} tasks - Interface for interacting with tasks.
 * @property {function(object): Promise<object>} tasks.insert - Inserts a new task.
 * @property {function(string): Promise<object>} tasks.get - Retrieves a task by ID.
 * @property {function(object): Promise<object>} tasks.list - Lists tasks with optional filters.
 * @property {function(string, object): Promise<object>} tasks.update - Updates a task by ID.
 * @property {object} steps - Interface for interacting with steps.
 * @property {function(object): Promise<object>} steps.insert - Inserts a new step.
 * @property {function(string): Promise<object>} steps.listByTask - Lists steps for a given task ID.
 * @property {function(string, object): Promise<object>} steps.update - Updates a step by ID.
 * @property {function(): object} getCacheContents - Returns the current contents of all in-memory caches.
 * @property {function(): void} clearCache - Clears all in-memory caches.
 */

/**
 * Initializes the Supabase storage client and returns an object with methods for interacting with various COP tables.
 * @param {object} [options] - Configuration options for the storage client.
 * @param {object} [options.supabaseClient] - An optional pre-initialized Supabase client. If provided, `supabaseUrl` and `supabaseServiceKey` are ignored.
 * @param {string} [options.supabaseUrl] - The URL of the Supabase project. Required if `supabaseClient` is not provided.
 * @param {string} [options.supabaseServiceKey] - The service role key for the Supabase project. Required if `supabaseClient` is not provided.
 * @param {('none'|'reads'|'writes'|'all')} [options.syncMode='none'] - Forces direct Supabase interaction, bypassing cache for debugging. 'none' (default) uses cache, 'reads' bypasses cache for reads, 'writes' bypasses cache for writes, 'all' bypasses cache for all operations.
 * @param {boolean} [options.debugMode=false] - Enables debug logging for storage operations.
 * @returns {object} An object containing methods for interacting with debugLogs, events, artifacts, agentIdentities, tasks, and steps, along with `getCacheContents()` and `clearCache()` for debugging.
 * @throws {Error} If `supabaseUrl` or `supabaseServiceKey` are not set and no `supabaseClient` is provided.
 */
/**
 * Creates an in-memory storage implementation that conforms to the StorageInterface.
 * This is useful for testing or when a persistent database is not required.
 * @returns {StorageInterface} An in-memory storage object.
 */
export function createInMemoryStorage() {
  const inMemoryData = {
    debugLogs: [],
    events: [],
    artifacts: [],
    agentIdentities: new Map(),
    tasks: new Map(),
    steps: new Map(),
  };

  const inMemoryStorage = {
    debugLogs: {
      async insert(logRecord) {
        inMemoryData.debugLogs.push(logRecord);
        return { ok: true };
      },
    },
    events: {
      async insert(eventRecord) {
        inMemoryData.events.push(eventRecord);
        return { ok: true, event: eventRecord };
      },
    },
    artifacts: {
      async insert(artifactRecord) {
        inMemoryData.artifacts.push(artifactRecord);
        return { ok: true, artifact: artifactRecord };
      },
    },
    agentIdentities: {
      async upsert(identity, conflictKey = "agent_name") {
        const existing = Array.from(inMemoryData.agentIdentities.values()).find(
          (a) => a[conflictKey] === identity[conflictKey]
        );
        if (existing) {
          Object.assign(existing, identity);
          inMemoryData.agentIdentities.set(existing.agent_id, existing);
          return { ok: true, identity: existing };
        } else {
          const newIdentity = { ...identity, agent_id: identity.agent_id || `agent_${inMemoryData.agentIdentities.size + 1}` };
          inMemoryData.agentIdentities.set(newIdentity.agent_id, newIdentity);
          return { ok: true, identity: newIdentity };
        }
      },
      async getById(agent_id) {
        const identity = inMemoryData.agentIdentities.get(agent_id);
        return { ok: !!identity, identity: identity || null };
      },
      async getByName(agent_name) {
        const identity = Array.from(inMemoryData.agentIdentities.values()).find(
          (a) => a.agent_name === agent_name
        );
        return { ok: !!identity, identity: identity || null };
      },
      async list({ status, limit = 100 } = {}) {
        let identities = Array.from(inMemoryData.agentIdentities.values());
        if (status) {
          identities = identities.filter((a) => a.status === status);
        }
        return { ok: true, identities: identities.slice(0, limit) };
      },
      async updateStatus(agent_id, status) {
        const identity = inMemoryData.agentIdentities.get(agent_id);
        if (identity) {
          identity.status = status;
          return { ok: true, identity };
        }
        return { ok: false, error: "Agent not found" };
      },
    },
    tasks: {
      async insert(taskRecord) {
        const newTask = { ...taskRecord, id: taskRecord.id || `task_${inMemoryData.tasks.size + 1}` };
        inMemoryData.tasks.set(newTask.id, newTask);
        return { ok: true, task: newTask };
      },
      async get(taskId) {
        const task = inMemoryData.tasks.get(taskId);
        return { ok: !!task, task: task || null };
      },
      async list({ status, limit = 100 } = {}) {
        let tasks = Array.from(inMemoryData.tasks.values());
        if (status) {
          tasks = tasks.filter((j) => j.status === status);
        }
        return { ok: true, tasks: tasks.slice(0, limit) };
      },
      async update(taskId, patch) {
        const task = inMemoryData.tasks.get(taskId);
        if (task) {
          Object.assign(task, patch);
          return { ok: true, task };
        }
        return { ok: false, error: "Task not found" };
      },
    },
    steps: {
      async insert(stepRecord) {
        const newStep = { ...stepRecord, id: stepRecord.id || `step_${inMemoryData.steps.size + 1}` };
        inMemoryData.steps.set(newStep.id, newStep);
        return { ok: true, step: newStep };
      },
      async listByTask(taskId) {
        const steps = Array.from(inMemoryData.steps.values()).filter(
          (s) => s.task_id === taskId
        );
        return { ok: true, steps };
      },
      async update(stepId, patch) {
        const step = inMemoryData.steps.get(stepId);
        if (step) {
          Object.assign(step, patch);
          return { ok: true, step };
        }
        return { ok: false, error: "Step not found" };
      },
    },
    getCacheContents: () => ({
      agentIdentities: Array.from(inMemoryData.agentIdentities.entries()),
      tasks: Array.from(inMemoryData.tasks.entries()),
      steps: Array.from(inMemoryData.steps.entries()),
    }),
    clearCache: () => {
      inMemoryData.agentIdentities.clear();
      inMemoryData.tasks.clear();
      inMemoryData.steps.clear();
      inMemoryData.debugLogs = [];
      inMemoryData.events = [];
      inMemoryData.artifacts = [];
    },
  };

  return inMemoryStorage;
}

export function initStorage( { syncMode = 'none', debugMode = false, ...options } = {} ) {
  const opt_client = options.supabaseClient;
  const url = options.supabaseUrl;
  const key = options.supabaseServiceKey;
  let currentSyncMode = syncMode;
    let currentDebugMode = debugMode;

    const logOperation = (type, operation, details = {}) => {
      if (currentDebugMode) {
        console.log(`[Storage Debug] ${type}: ${operation}`, details);
      }
    };
  if( !opt_client ){
    if (!url || !key) {
      throw new Error("COP.createStorage: supabaseUrl or supabaseServiceKey not set");
    }
  }
  const client = opt_client || createClient(url, key);

  // Caches for frequently accessed data
  /**
   * @private
   * @type {Map<string, object>}
   * @description Cache for agent identity records, keyed by agent_id or parameterized list keys.
   */
  const agentIdentitiesCache = new Map();
  /**
   * @private
   * @type {Map<string, object>}
   * @description Cache for task records, keyed by task ID or parameterized list keys.
   */
  const tasksCache = new Map();
  /**
   * @private
   * @type {Map<string, object>}
   * @description Cache for step records, keyed by task ID.
   */
  const stepsCache = new Map();

  const flushAllCaches = () => {
    agentIdentitiesCache.clear();
    tasksCache.clear();
    stepsCache.clear();
  };

  /**
   * Performs a deep comparison between two objects to determine if they are equal.
   * @param {object} obj1 - The first object to compare.
   * @param {object} obj2 - The second object to compare.
   * @returns {boolean} True if the objects are deeply equal, false otherwise.
   */
  const areObjectsEqual = (obj1, obj2) => {
    if (obj1 === obj2) return true;
    if (typeof obj1 !== 'object' || obj1 === null || typeof obj2 !== 'object' || obj2 === null) return false;

    const keys1 = Object.keys(obj1);
    const keys2 = Object.keys(obj2);

    if (keys1.length !== keys2.length) return false;

    for (const key of keys1) {
      if (!keys2.includes(key) || !areObjectsEqual(obj1[key], obj2[key])) {
        return false;
      }
    }
    return true;
  };

  const debugLogs = {
    /**
     * Inserts a debug log record into the 'cop_debug_logs' table.
     * @param {object} logRecord - The debug log record to insert.
     * @returns {Promise<object>} An object indicating success or failure, with an error message if applicable.
     */
    async insert(logRecord) {
      const { error } = await client.from("cop_debug_logs").insert(logRecord);
      if (error) {
        return { ok: false, error: error.message };
      }
      return { ok: true };
    },
  };

  const events = {
    /**
     * Inserts an event record into the 'cop_events' table.
     * @param {object} eventRecord - The event record to insert.
     * @returns {Promise<object>} An object indicating success or failure, with the inserted event data if successful.
     */
    async insert(eventRecord) {
      const { data, error } = await client
        .from("cop_events")
        .insert(eventRecord)
        .select()
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message };
      }
      return { ok: true, event: data };
    },
  };

  const artifacts = {
    /**
     * Inserts an artifact record into the 'cop_artifacts' table.
     * @param {object} artifactRecord - The artifact record to insert.
     * @returns {Promise<object>} An object indicating success or failure, with the inserted artifact data if successful.
     */
    async insert(artifactRecord) {
      const { data, error } = await client
        .from("cop_artifacts")
        .insert(artifactRecord)
        .select()
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message };
      }
      return { ok: true, artifact: data };
    },
  };

  const agentIdentities = {
    /**
     * Inserts or updates an agent identity record in the 'cop_agent_identities' table. Avoids duplicate writes if the record is unchanged. Invalidates relevant caches upon successful insertion/update.
     * @param {object} identity - The agent identity record to upsert.
     * @param {string} [conflictKey="agent_name"] - The column to use for conflict resolution.
     * @returns {Promise<object>} An object indicating success or failure, with the upserted identity data if successful.
     */
    async upsert(identity, conflictKey = "agent_name") {
      if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
        const { data, error } = await client
          .from("cop_agent_identities")
          .upsert(identity, { onConflict: conflictKey })
          .select()
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, identity: null };
        }
        logOperation('Write', 'agentIdentities.upsert', { agent_id: data.agent_id });
        flushAllCaches();
        return { ok: true, identity: data };
      }

      if (identity.agent_id && agentIdentitiesCache.has(identity.agent_id)) {
        const cachedIdentity = agentIdentitiesCache.get(identity.agent_id);
        if (areObjectsEqual(identity, cachedIdentity)) {
          return { ok: true, identity: cachedIdentity };
        }
      }

      const { data, error } = await client
        .from("cop_agent_identities")
        .upsert(identity, { onConflict: conflictKey })
        .select()
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message };
      }
      return { ok: true, identity: data };
    },
    async getById(agent_id) {
      /**
       * Retrieves an agent identity record by its ID, utilizing a cache.
     * @param {string} agent_id - The ID of the agent to retrieve.
     * @returns {Promise<object>} An object indicating success or failure, with the identity data if found.
     */
      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        const { data, error } = await client
          .from("cop_agent_identities")
          .select("*")
          .eq("agent_id", agent_id)
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, identity: null };
        }
        logOperation('Write', 'agentIdentities.updateStatus', { agent_id: data.agent_id, status: data.status });
        flushAllCaches();
        return { ok: true, identity: data };
      }

      if (agentIdentitiesCache.has(agent_id)) {
        logOperation('Read', 'agentIdentities.getById', { agent_id, cached: true });
        return { ok: true, identity: agentIdentitiesCache.get(agent_id) };
      }

      logOperation('Read', 'agentIdentities.getById', { agent_id, cached: false });
      const { data, error } = await client
        .from("cop_agent_identities")
        .select("*")
        .eq("agent_id", agent_id)
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message, identity: null };
      }
      if (data) {
        agentIdentitiesCache.set(agent_id, data);
      }
      return { ok: true, identity: data || null };
    },
    async getByName(agent_name) {
      /**
       * Retrieves an agent identity record by its name, utilizing a cache.
     * @param {string} agent_name - The name of the agent to retrieve.
     * @returns {Promise<object>} An object indicating success or failure, with the identity data if found.
     */
      // For getByName, we'll iterate the cache as agent_name is not the key
      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        const { data, error } = await client
          .from("cop_agent_identities")
          .select("*")
          .eq("agent_name", agent_name)
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, identity: null };
        }
        flushAllCaches();
        return { ok: true, identity: data };
      }

      for (const identity of agentIdentitiesCache.values()) {
        if (identity.agent_name === agent_name) {
          logOperation('Read', 'agentIdentities.getByName', { agent_name, cached: true });
          return { ok: true, identity };
        }
      }

      logOperation('Read', 'agentIdentities.getByName', { agent_name, cached: false });
      const { data, error } = await client
        .from("cop_agent_identities")
        .select("*")
        .eq("agent_name", agent_name)
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message, identity: null };
      }
      if (data) {
        agentIdentitiesCache.set(data.agent_id, data);
      }
      return { ok: true, identity: data || null };
    },
    async list({ status, limit = 100 } = {}) {
      /**
       * Lists agent identity records, optionally filtered by status and limited by count, utilizing a cache.
       * @param {object} [params] - Parameters for listing identities.
       * @param {string} [params.status] - Filter identities by status.
       * @param {number} [params.limit=100] - Maximum number of identities to return.
       * @returns {Promise<object>} An object indicating success or failure, with an array of identity data if successful.
       */
      const fetchFromDatabase = async () => {
        let query = client
          .from("cop_agent_identities")
          .select("*")
          .order("created_at", { ascending: true })
          .limit(limit);
        if (status) {
          query = query.eq("status", status);
        }
        const { data, error } = await query;
        if (error) {
          return { ok: false, error: error.message, identities: [] };
        }
        flushAllCaches();
        return { ok: true, identities: data || [] };
      };

      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        return fetchFromDatabase();
      }

      const cacheKey = `list-${status || 'all'}-${limit}`;
      if (agentIdentitiesCache.has(cacheKey)) {
            logOperation('Read', 'agentIdentities.list', { status, limit, cached: true });
            return { ok: true, identities: agentIdentitiesCache.get(cacheKey) };
          }

          logOperation('Read', 'agentIdentities.list', { status, limit, cached: false });

      const result = await fetchFromDatabase();
      if (result.ok) {
        agentIdentitiesCache.set(cacheKey, result.identities);
      }
      return result;
    },
    async updateStatus(agent_id, status) {
      /**
       * Updates the status of an agent identity in the 'cop_agent_identities' table. Avoids duplicate writes if the status is unchanged. Invalidates relevant caches upon successful update.
     * @param {string} agent_id - The ID of the agent to update.
     * @param {string} status - The new status for the agent.
     * @returns {Promise<object>} An object indicating success or failure, with the updated identity data if successful.
     */
      if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
        const { data, error } = await client
          .from("cop_agent_identities")
          .update({ status })
          .eq("agent_id", agent_id)
          .select()
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, identity: null };
        }
        flushAllCaches();
        return { ok: true, identity: data };
      }

      if (agentIdentitiesCache.has(agent_id)) {
        const cachedIdentity = agentIdentitiesCache.get(agent_id);
        const potentialNewIdentity = { ...cachedIdentity, status };
        if (areObjectsEqual(potentialNewIdentity, cachedIdentity)) {
          return { ok: true, identity: cachedIdentity };
        }
      }

      const { data, error } = await client
        .from("cop_agent_identities")
        .update({ status })
        .eq("agent_id", agent_id)
        .select()
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message, identity: null };
      }
      if (data) {
        // Invalidate cache for the upserted identity
        agentIdentitiesCache.delete(data.agent_id);
        agentIdentitiesCache.delete('list'); // Invalidate the list cache
      }
      return { ok: true, identity: data || null };
    },
  };

  const tasks = {
    /**
     * Inserts or updates a task record in the 'cop_tasks' table. Avoids duplicate writes if the record is unchanged. Invalidates relevant caches upon successful insertion/update.
     * @param {object} taskRecord - The task record to insert or update.
     * @returns {Promise<object>} An object indicating success or failure, with the inserted/updated task data if successful.
     */
    async insert(taskRecord) {
      if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
        const { data, error } = await client
          .from("cop_tasks")
          .upsert(taskRecord, { onConflict: 'id' })
          .select()
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, task: null };
        }
        logOperation('Write', 'tasks.insert', { task_id: data.id });
        flushAllCaches();
        return { ok: true, task: data };
      }

      if (tasksCache.has(taskRecord.id)) {
        const cachedTask = tasksCache.get(taskRecord.id);
        if (areObjectsEqual(taskRecord, cachedTask)) {
          return { ok: true, task: cachedTask };
        }
      }

      const { data, error } = await client
        .from("cop_tasks")
        .upsert(taskRecord, { onConflict: 'id' })
        .select()
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message, task: null };
      }
      if (data) {
        tasksCache.delete(data.id);
        for (const key of tasksCache.keys()) {
          if (key.startsWith('list-')) {
            tasksCache.delete(key);
          }
        }
      }
      return { ok: true, task: data };
    },
    async update(taskId, patch) {
      /**
       * Updates a task record in the 'cop_tasks' table. Avoids duplicate writes if the patch would not change the record. Invalidates relevant caches upon successful update.
     * @param {string} taskId - The ID of the task to update.
     * @param {object} patch - The partial object containing fields to update.
     * @returns {Promise<object>} An object indicating success or failure, with the updated task data if successful.
     */
      if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
        const { data, error } = await client
          .from("cop_tasks")
          .update(patch)
          .eq("id", taskId)
          .select()
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, task: null };
        }
        logOperation('Write', 'tasks.update', { task_id: data.id, status: data.status });
        flushAllCaches();
        return { ok: true, task: data };
      }

      if (tasksCache.has(taskId)) {
        const cachedTask = tasksCache.get(taskId);
        const potentialNewTask = { ...cachedTask, ...patch };
        if (areObjectsEqual(potentialNewTask, cachedTask)) {
          return { ok: true, task: cachedTask };
        }
      }

      const { data, error } = await client
        .from("cop_tasks")
        .update(patch)
        .eq("id", taskId)
        .select()
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message, task: null };
      }
      if (data) {
        tasksCache.delete(taskId);
        for (const key of tasksCache.keys()) {
          if (key.startsWith('list-')) {
            tasksCache.delete(key);
          }
        }
      }
      return { ok: true, task: data };
    },
    async get(taskId) {
      /**
       * Retrieves a task record by its ID, utilizing a cache.
     * @param {string} taskId - The ID of the task to retrieve.
     * @returns {Promise<object>} An object indicating success or failure, with the task data if found.
     */
      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        const { data, error } = await client
          .from("cop_tasks")
          .select("*")
          .eq("id", taskId)
          .maybeSingle();
        if (error) {
          return { ok: false, error: error.message, task: null };
        }
        flushAllCaches();
        return { ok: true, task: data };
      }

      if (tasksCache.has(taskId)) {
        logOperation('Read', 'tasks.get', { taskId, cached: true });
        return { ok: true, task: tasksCache.get(taskId) };
      }

      logOperation('Read', 'tasks.get', { taskId, cached: false });

      const { data, error } = await client
        .from("cop_tasks")
        .select("*")
        .eq("id", taskId)
        .maybeSingle();
      if (error) {
        return { ok: false, error: error.message, task: null };
      }
      if (data) {
        tasksCache.set(taskId, data);
      }
      return { ok: true, task: data || null };
    },
    async list(params = {}) {
      /**
       * Lists task records, optionally filtered by status, task type, worker agent name, and limited by count, utilizing a cache.
     * @param {object} [params] - Parameters for listing tasks.
     * @param {string} [params.status] - Filter tasks by status.
     * @param {string} [params.task_type] - Filter tasks by type.
     * @param {string} [params.worker_agent_name] - Filter tasks by worker agent name.
     * @param {number} [params.limit=100] - Maximum number of tasks to return.
     * @returns {Promise<object>} An object indicating success or failure, with an array of task data if successful.
     */
      const { status, task_type, worker_agent_name, limit = 100 } = params;

      const fetchFromDatabase = async () => {
        let q = client
          .from("cop_tasks")
          .select("*")
          .order("created_at", { ascending: false })
          .limit(limit);

        if (status) {
          q = q.eq("status", status);
        }
        if (task_type) {

export function getStorage(options) {
  /**
   * Retrieves a Supabase storage instance.
   *
   * This function prioritizes an explicitly provided storage instance, then the global default,
   * and finally initializes a new one if no suitable existing instance is found or if
   * provided options conflict with existing configurations.
   *
   * @param {object} [options] - Configuration options for the storage instance.
   * @param {object} [options.storage] - An existing storage instance to potentially use.
   *                                     If provided and its configuration (supabaseUrl/Key)
   *                                     matches the other `options` properties, it will be returned.
   * @param {string} [options.supabaseUrl] - The Supabase project URL. If provided, it must match
   *                                         the URL of any existing storage instance being considered.
   * @param {string} [options.supabaseServiceKey] - The Supabase service role key. If provided, it must match
   *                                                the key of any existing storage instance being considered.
   * @returns {object} The appropriate storage instance, either an existing one or a newly initialized one.
   *
   * @remarks
   * - If called without `options` (e.g., `getStorage()`), it returns the current `defaultStorage`.
   * - If `options.storage` is provided and its `supabaseUrl` and `supabaseServiceKey` (if specified in `options`)
   *   match, that `options.storage` instance is returned.
   * - If `defaultStorage` exists and its `supabaseUrl` and `supabaseServiceKey` (if specified in `options`)
   *   match, that `defaultStorage` instance is returned.
   * - Otherwise, a new storage instance is initialized using the provided `options` and set as the new `defaultStorage`.
   */

  const providedStorage = options?.storage;

  // Helper to check if a storage instance's configuration matches the current request options
  const configMatches = (storageInstance) => {
    // If no options are provided to getStorage(), any existing storage is considered a match for "no specific config".
    if (!options) {
      return true;
    }
    // If the storageInstance itself or its options are missing, it cannot match the provided options.
    if (!storageInstance || !storageInstance.options) {
      return false;
    }

    let matches = true;

    // If options.supabaseUrl is provided, it must match storageInstance.options.supabaseUrl
    if (options.supabaseUrl !== undefined) {
      if (options.supabaseUrl !== storageInstance.options.supabaseUrl) {
        matches = false;
      }
    }

    // If options.supabaseServiceKey is provided, it must match storageInstance.options.supabaseServiceKey
    if (options.supabaseServiceKey !== undefined) {
      if (options.supabaseServiceKey !== storageInstance.options.supabaseServiceKey) {
        matches = false;
      }
    }

    return matches;
  };

  // 1. Check if a storage instance is explicitly provided in options and if its config matches
  if (providedStorage && configMatches(providedStorage)) {
    // If it matches, use it and update defaultStorage if necessary
    if (providedStorage !== defaultStorage) {
      defaultStorage = providedStorage;
    }
    return providedStorage;
  }

  // 2. Check if defaultStorage exists and if its config matches
  if (defaultStorage && configMatches(defaultStorage)) {
    return defaultStorage;
  }

  // 3. If neither of the above, initialize a new storage instance
  defaultStorage = initStorage(options);
  return defaultStorage;
}

export function setStorage(storage) {
  /**
   * Sets the default storage instance.
   * @param {object} storage - The storage instance to set as default.
   */
  defaultStorage = storage;
}

/**
 * Creates an in-memory storage implementation that conforms to the StorageInterface.
 * This is useful for testing or when a persistent database is not required.

export function createInMemoryStorage() {
  const inMemoryData = {
    debugLogs: [],
    events: [],
    artifacts: [],
    agentIdentities: new Map(),
    tasks: new Map(),
    steps: new Map(),
  };

  // --- MISE À JOUR : Les méthodes `insert` sont renommées en `upsert` dans l'implémentation en mémoire pour coller à l'interface réelle ---

  const inMemoryStorage = {
    debugLogs: {
      async insert(logRecord) {
        inMemoryData.debugLogs.push(logRecord);
        return { ok: true };
      },
    },
    events: {
      async insert(eventRecord) {
        inMemoryData.events.push(eventRecord);
        return { ok: true, event: eventRecord };
      },
    },
    artifacts: {
      async insert(artifactRecord) {
        inMemoryData.artifacts.push(artifactRecord);
        return { ok: true, artifact: artifactRecord };
      },
    },
    agentIdentities: {
      async upsert(identity, conflictKey = "agent_name") {
        const existing = Array.from(inMemoryData.agentIdentities.values()).find(
          (a) => a[conflictKey] === identity[conflictKey]
        );
        if (existing) {
          Object.assign(existing, identity);
          inMemoryData.agentIdentities.set(existing.agent_id, existing);
          return { ok: true, identity: existing };
        } else {
          const newIdentity = { ...identity, agent_id: identity.agent_id || `agent_${inMemoryData.agentIdentities.size + 1}` };
          inMemoryData.agentIdentities.set(newIdentity.agent_id, newIdentity);
          return { ok: true, identity: newIdentity };
        }
      },
      async getById(agent_id) {
        const identity = inMemoryData.agentIdentities.get(agent_id);
        return { ok: !!identity, identity: identity || null };
      },
      async getByName(agent_name) {
        const identity = Array.from(inMemoryData.agentIdentities.values()).find(
          (a) => a.agent_name === agent_name
        );
        return { ok: !!identity, identity: identity || null };
      },
      async list({ status, limit = 100 } = {}) {
        let identities = Array.from(inMemoryData.agentIdentities.values());
        if (status) {
          identities = identities.filter((a) => a.status === status);
        }
        return { ok: true, identities: identities.slice(0, limit) };
      },
      async updateStatus(agent_id, status) {
        const identity = inMemoryData.agentIdentities.get(agent_id);
        if (identity) {
          identity.status = status;
          return { ok: true, identity };
        }
        return { ok: false, error: "Agent not found", code: ERROR_CODES.NOT_FOUND };
      },
    },
    tasks: {
      async upsert(taskRecord) {
        const newTask = { ...taskRecord, id: taskRecord.id || `task_${inMemoryData.tasks.size + 1}` };
        inMemoryData.tasks.set(newTask.id, newTask);
        return { ok: true, task: newTask };
      },
      async get(taskId) {
        const task = inMemoryData.tasks.get(taskId);
        return { ok: !!task, task: task || null };
      },
      async list({ status, limit = 100 } = {}) {
        let tasks = Array.from(inMemoryData.tasks.values());
        if (status) {
          tasks = tasks.filter((j) => j.status === status);
        }
        return { ok: true, tasks: tasks.slice(0, limit) };
      },
      async update(taskId, patch) {
        const task = inMemoryData.tasks.get(taskId);
        if (task) {
          Object.assign(task, patch);
          return { ok: true, task };
        }
        return { ok: false, error: "Task not found", code: ERROR_CODES.NOT_FOUND };
      },
    },
    steps: {

        inMemoryData.steps.set(newStep.id, newStep);
        return { ok: true, step: newStep };
      },
      async listByTask(taskId) {
        const steps = Array.from(inMemoryData.steps.values()).filter(
          (s) => s.task_id === taskId
        );
        return { ok: true, steps };
      },
      async update(taskId, stepId, patch) { // <-- CORRECTION: Ajout de taskId pour coller à l'interface
        const step = inMemoryData.steps.get(stepId);
        if (step && step.task_id === taskId) {
          Object.assign(step, patch);
          return { ok: true, step };
        }
        return { ok: false, error: "Step not found", code: ERROR_CODES.NOT_FOUND };
      },
    },
    getCacheContents: () => ({

export function initStorage( { syncMode = 'none', debugMode = false, ...options } = {} ) {
  const opt_client = options.supabaseClient;
  const url = options.supabaseUrl;
  const key = options.supabaseServiceKey;
  let currentSyncMode = syncMode;
  let currentDebugMode = debugMode;

  const logOperation = (type, operation, details = {}) => {
    if (currentDebugMode) {
      console.log(`[Storage Debug] ${type}: ${operation}`, details);
    }
  };
  if( !opt_client ){
    if (!url || !key) {
      throw new Error("COP.createStorage: supabaseUrl or supabaseServiceKey not set");
    }
  }
  const client = opt_client || createClient(url, key);

  // Caches for frequently accessed data
  const agentIdentitiesCache = new Map();
  const tasksCache = new Map();
  const stepsCache = new Map();

  const flushAllCaches = () => {
    agentIdentitiesCache.clear();
    tasksCache.clear();
    stepsCache.clear();
  };

  /**
   * Performs a deep comparison between two objects to determine if they are equal.
   * @param {object} obj1 - The first object to compare.
   * @param {object} obj2 - The second object to compare.
   * @returns {boolean} True if the objects are deeply equal, false otherwise.
   */
  const areObjectsEqual = (obj1, obj2) => {
    if (obj1 === obj2) return true;
    if (typeof obj1 !== 'object' || obj1 === null || typeof obj2 !== 'object' || obj2 === null) return false;

    const keys1 = Object.keys(obj1);
    const keys2 = Object.keys(obj2);

    if (keys1.length !== keys2.length) return false;

    for (const key of keys1) {
      if (!keys2.includes(key) || !areObjectsEqual(obj1[key], obj2[key])) {
        return false;
      }
    }
    return true;
  };

  // --- DÉFINITION DE L'INTERFACE DE LOGGING D'ERREUR ---
  const debugLogs = {
    /**
     * Inserts a debug log record into the 'cop_debug_logs' table.
     * @param {object} logRecord - The debug log record to insert.
     * @returns {Promise<object>} An object indicating success or failure, with an error message if applicable.
     */
    async insert(logRecord) {
      // Évite la récursivité infinie en cas d'erreur lors de l'insertion du log
      try {
        const { error } = await client.from("cop_debug_logs").insert(logRecord);
        if (error) {
          // Log console de dernier recours si l'insertion DB échoue
          console.error("[FATAL LOG ERROR] Failed to insert debug log:", error.message, logRecord);
          return { ok: false, error: error.message };
        }
        return { ok: true };
      } catch (e) {
        console.error("[CRITICAL LOG ERROR] Exception during debug log insertion:", e.message, logRecord);
        return { ok: false, error: e.message };
      }
    },
  };

  // --- FONCTION UTILITAIRE GÉNÉRIQUE POUR EXÉCUTER LES OPÉRATIONS DE STOCKAGE ---

  /**
   * Executes a storage operation, handling syncMode, logging, and error insertion.
   * Cette fonction implémente le logging d'erreur et le flushing de cache après une écriture réussie.
   * @param {string} type - 'Read' or 'Write'.
   * @param {string} operation - Name of the operation (e.g., 'agentIdentities.getById').
   * @param {function(): Promise<object>} dbOperation - The function to execute the Supabase query, returning { data, error, count, status }.
   * @param {object} [details={}] - Additional details for logging.
   * @returns {Promise<object>} The result object { ok, data (or null), error (or null), code (or null) }.
   */
  const executeStorageOperation = async (type, operation, dbOperation, details = {}) => {
    try {
      logOperation(type, operation, details);
      const { data, error, status } = await dbOperation();

      if (error) {
        // Log l'erreur dans la table debugLogs
        await debugLogs.insert({
          level: 'error',
          message: `[DB Error] ${operation} failed: ${error.message}`,
          context: {
            error: { code: error.code, message: error.message, hint: error.hint },
            details,
            status,
          },
        });
        // Retourne l'erreur standardisée
        return {
          ok: false,
          error: error.message,
          code: ERROR_CODES.DB_ERROR,
        };
      }

      if (type === 'Write') {
        // Invalider tous les caches pour garantir la cohérence après une écriture
        flushAllCaches();
      }

      return { ok: true, data: data };
    } catch (e) {
      // Log les erreurs d'exécution inattendues (ex: erreur de code, client non initialisé)
      await debugLogs.insert({
        level: 'fatal',
        message: `[Fatal Exception] ${operation} execution failed: ${e.message}`,
        context: { error: e.stack, details },
      });
      return {
        ok: false,
        error: e.message,
        code: ERROR_CODES.DB_ERROR, // Utilise DB_ERROR pour les échecs bas niveau
      };
    }
  };

  // --- FONCTIONS CRUD GÉNÉRIQUES POUR L'USAGE INTERNE ---

  /**
   * Gère la logique de cache de lecture pour les enregistrements individuels par ID.
   */
  const handleReadSingle = async (tableName, id, cache, operationName, idKey = 'id') => {
    // 1. SyncMode Bypassing
    if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
      const dbResult = await executeStorageOperation('Read', operationName, () =>
        client.from(tableName).select("*").eq(idKey, id).maybeSingle()
      );
      return { ...dbResult, [idKey]: dbResult.data };
    }

    // 2. Cache Lookup
    if (cache.has(id)) {
      logOperation('Read', operationName, { id, cached: true });
      return { ok: true, [idKey]: cache.get(id) };
    }

    // 3. Database Fetch
    const dbResult = await executeStorageOperation('Read', operationName, () =>
      client.from(tableName).select("*").eq(idKey, id).maybeSingle()
    );

    // 4. Cache Populate
    if (dbResult.ok && dbResult.data) {
      cache.set(id, dbResult.data);
    }

    // Renvoie l'objet avec la clé appropriée pour l'ID
    return { ...dbResult, [idKey]: dbResult.data || null };
  };

  /**
   * Gère la logique d'upsert (insertion ou mise à jour) avec vérification de l'égalité et cache-bypassing.
   * @param {string} tableName - Nom de la table.
   * @param {object} record - L'enregistrement complet ou partiel à upsert.
   * @param {Map<string, object>} cache - Le cache de la table (keyed par ID).
   * @param {string} operationName - Nom de l'opération pour le logging.
   * @param {string} idKey - Clé d'identification de l'enregistrement (ex: 'agent_id', 'id').
   * @param {string} conflictKey - Clé de résolution de conflit pour Supabase.
   * @returns {Promise<object>} Le résultat de l'opération.
   */
  const handleUpsert = async (tableName, record, cache, operationName, idKey, conflictKey = idKey) => {
    // 1. SyncMode Bypassing (écrit directement)
    if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
      const dbResult = await executeStorageOperation('Write', operationName, () =>
        client.from(tableName).upsert(record, { onConflict: conflictKey }).select().maybeSingle()
      );
      return { ...dbResult, [idKey]: dbResult.data };
    }

    // 2. Vérification du cache pour éviter une écriture inutile
    const recordId = record[idKey];
    if (recordId && cache.has(recordId)) {
      const cachedRecord = cache.get(recordId);
      // Vérification que les données sont identiques, mais seulement si le record contient tous les champs de l'objet en cache
      // Note: Pour un upsert, on assume que 'record' est complet, ou suffisant pour une égalité.
      if (areObjectsEqual(record, cachedRecord)) {
        logOperation('Write', operationName, { id: recordId, skipped: true });
        return { ok: true, [idKey]: cachedRecord };
      }
    }

    // 3. Database Write
    const dbResult = await executeStorageOperation('Write', operationName, () =>
      client.from(tableName).upsert(record, { onConflict: conflictKey }).select().maybeSingle()
    );

    // 4. Invalidation ciblée (le flushAllCaches est déjà fait dans executeStorageOperation, mais on le laisse pour l'exemple)
    if (dbResult.ok && dbResult.data) {
      cache.delete(dbResult.data[idKey]); // Supprimer l'ancienne entrée avant le prochain fetch
    }

    return { ...dbResult, [idKey]: dbResult.data };
  };

  /**
   * Gère la logique de mise à jour partielle avec gestion du cache et du versionnement optimiste.
   * @param {string} tableName - Nom de la table.
   * @param {string} id - L'ID de l'enregistrement à mettre à jour.
   * @param {object} patch - Le patch (champs à mettre à jour).
   * @param {Map<string, object>} cache - Le cache de la table (keyed par ID).
   * @param {string} operationName - Nom de l'opération pour le logging.
   * @param {string} idKey - Clé d'identification de l'enregistrement.
   * @param {string} [versionField] - Nom du champ de version pour le versionnement optimiste (si présent dans le patch).
   * @returns {Promise<object>} Le résultat de l'opération.
   */
  const handleUpdate = async (tableName, id, patch, cache, operationName, idKey, versionField = 'version') => {
    // Vérifie si le versionnement optimiste est demandé dans le patch
    const useOptimisticLock = versionField in patch;
    let query = client.from(tableName).update(patch).eq(idKey, id).select().maybeSingle();

    if (useOptimisticLock) {
      const expectedVersion = patch[versionField];
      // Pour éviter les collisions, on utilise la version actuelle dans la clause WHERE
      query = client.from(tableName)
        .update({ ...patch, [versionField]: expectedVersion + 1 }) // Incrémente la version côté DB
        .eq(idKey, id)
        .eq(versionField, expectedVersion) // Condition de verrouillage optimiste
        .select()
        .maybeSingle();
      logOperation('Write', operationName, { id, optimisticLock: true, expectedVersion });
    } else {
        logOperation('Write', operationName, { id, optimisticLock: false });
    }

    // 1. SyncMode Bypassing & Cache Check (Optimisation de l'écriture inutile)
    if (currentSyncMode !== 'all' && currentSyncMode !== 'writes') {
      if (cache.has(id)) {
        const cachedRecord = cache.get(id);
        const potentialNewRecord = { ...cachedRecord, ...patch };
        // Vérifie si le patch entraînerait un changement
        if (areObjectsEqual(potentialNewRecord, cachedRecord)) {
          logOperation('Write', operationName, { id, skipped: true });
          return { ok: true, [idKey]: cachedRecord };
        }
      }
    }


    // 2. Database Write
    const dbResult = await executeStorageOperation('Write', operationName, () => query);

    if (dbResult.ok && dbResult.data) {
      cache.delete(id); // Invalide l'entrée individuelle
    } else if (!dbResult.ok && useOptimisticLock && dbResult.error?.includes('0 rows')) {
        // Gère spécifiquement l'échec du verrouillage optimiste (pas de ligne mise à jour)
        return {
            ok: false,
            error: "Optimistic lock failed. Record version mismatch.",
            code: ERROR_CODES.OPTIMISTIC_LOCK_FAIL,
        };
    }

    return { ...dbResult, [idKey]: dbResult.data };
  };

  const events = {
    /**
     * Inserts an event record into the 'cop_events' table.
     */
    async insert(eventRecord) {
      return executeStorageOperation('Write', 'events.insert', () =>
        client.from("cop_events").insert(eventRecord).select().maybeSingle()
      );
    },
  };

  const artifacts = {
    /**
     * Inserts an artifact record into the 'cop_artifacts' table.
     */
    async insert(artifactRecord) {
      return executeStorageOperation('Write', 'artifacts.insert', () =>
        client.from("cop_artifacts").insert(artifactRecord).select().maybeSingle()
      );
    },
  };

  const agentIdentities = {
    /**
     * Inserts or updates an agent identity record in the 'cop_agent_identities' table.
     */
    async upsert(identity, conflictKey = "agent_name") {
      return handleUpsert(
        "cop_agent_identities",
        identity,
        agentIdentitiesCache,
        'agentIdentities.upsert',
        'agent_id',
        conflictKey
      );
    },
    async getById(agent_id) {
      /**
       * Retrieves an agent identity record by its ID, utilizing a cache.
       */
      return handleReadSingle(
        "cop_agent_identities",
        agent_id,
        agentIdentitiesCache,
        'agentIdentities.getById',
        'agent_id'
      );
    },
    async getByName(agent_name) {
      /**
       * Retrieves an agent identity record by its name, utilizing a cache.
       * Note: Utilise la BDD car la clé n'est pas l'ID de cache.
       */
      return handleReadSingle(
        "cop_agent_identities",
        agent_name,
        agentIdentitiesCache,
        'agentIdentities.getByName',
        'agent_id' // Renvoie la clé 'identity'
      );
    },
    async list({ status, limit = 100 } = {}) {
      /**
       * Lists agent identity records, utilizes a cache keyed by list parameters.
       */
      const fetchFromDatabase = async () => {
        let query = client
          .from("cop_agent_identities")
          .select("*")
          .order("created_at", { ascending: true })
          .limit(limit);
        if (status) {
          query = query.eq("status", status);
        }
        return query;
      };

      const cacheKey = `list-identities-${status || 'all'}-${limit}`;

      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        const dbResult = await executeStorageOperation('Read', 'agentIdentities.list', fetchFromDatabase);
        return { ...dbResult, identities: dbResult.data || [] };
      }

      if (agentIdentitiesCache.has(cacheKey)) {
        logOperation('Read', 'agentIdentities.list', { status, limit, cached: true });
        return { ok: true, identities: agentIdentitiesCache.get(cacheKey) };
      }

      const result = await executeStorageOperation('Read', 'agentIdentities.list', fetchFromDatabase);
      if (result.ok) {
        agentIdentitiesCache.set(cacheKey, result.data);
      }
      return { ...result, identities: result.data || [] };
    },
    async updateStatus(agent_id, status) {
      /**
       * Updates the status of an agent identity in the 'cop_agent_identities' table.
       */
      const patch = { status };
      // Note: On ne supporte pas le verrouillage optimiste par défaut sur updateStatus
      return handleUpdate(
        "cop_agent_identities",
        agent_id,
        patch,
        agentIdentitiesCache,
        'agentIdentities.updateStatus',
        'agent_id'
      );
    },
  };

  const tasks = {
    /**
     * Inserts or updates a task record in the 'cop_tasks' table. Renommé de 'insert' à 'upsert'.
     */
    async upsert(taskRecord) {
      return handleUpsert(
        "cop_tasks",
        taskRecord,
        tasksCache,
        'tasks.upsert',
        'id',
        'id'
      );
    },
    async update(taskId, patch) {
      /**
       * Updates a task record in the 'cop_tasks' table. Supporte le versionnement optimiste si `patch.version` est fourni.
       */
      return handleUpdate(
        "cop_tasks",
        taskId,
        patch,
        tasksCache,
        'tasks.update',
        'id',
        'version'
      );
    },
    async get(taskId) {
      /**
       * Retrieves a task record by its ID, utilizing a cache.
       */
      return handleReadSingle(
        "cop_tasks",
        taskId,
        tasksCache,
        'tasks.get',
        'task'
      );
    },
    async list(params = {}) {
      /**
       * Lists task records, utilizes a cache keyed by list parameters.
       */
      const { status, task_type, worker_agent_name, limit = 100 } = params;

      const fetchFromDatabase = async () => {
        let q = client
          .from("cop_tasks")
          .select("*")
          .order("created_at", { ascending: false })
          .limit(limit);

        if (status) {
          q = q.eq("status", status);
        }
        if (task_type) {
          q = q.eq("task_type", task_type);
        }
        if (worker_agent_name) {
          q = q.eq("worker_agent_name", worker_agent_name);
        }

        return q;
      };

      const cacheKey = `list-tasks-${status || 'all'}-${task_type || 'all'}-${worker_agent_name || 'all'}-${limit}`;

      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        const dbResult = await executeStorageOperation('Read', 'tasks.list', fetchFromDatabase);
        return { ...dbResult, tasks: dbResult.data || [] };
      }

      if (tasksCache.has(cacheKey)) {
        logOperation('Read', 'tasks.list', { status, task_type, worker_agent_name, limit, cached: true });
        return { ok: true, tasks: tasksCache.get(cacheKey) };
      }

      const result = await executeStorageOperation('Read', 'tasks.list', fetchFromDatabase);
      if (result.ok) {
        tasksCache.set(cacheKey, result.data);
      }
      return { ...result, tasks: result.data || [] };
    },
  };

  const steps = {
    /**
     * Inserts or updates a step record into the 'cop_steps' table. Renommé de 'insert' à 'upsert'.
     */
    async upsert(stepRecord) {
      return handleUpsert(
        "cop_steps",
        stepRecord,
        stepsCache, // Cache des steps est keyed par task_id, mais on passe le map global ici.
        'steps.upsert',
        'id',
        'id'
      );
    },
    async update(taskId, stepId, patch) {
      /**
       * Updates a step record in the 'cop_steps' table.
       * @param {string} taskId - The ID of the task associated with the step (for cache lookup).
       * @param {string} stepId - The ID of the step to update.
       * @param {object} patch - The partial object containing fields to update.
       * @returns {Promise<object>} An object indicating success or failure, with the updated step data if successful.
       */

      // La logique de vérification du cache doit se faire ici car le cache stepsCache est keyed par task_id.

      // 1. Check for write bypass using cache
      if (currentSyncMode !== 'all' && currentSyncMode !== 'writes' && stepsCache.has(taskId)) {
        const cachedSteps = stepsCache.get(taskId);
        const existingStep = cachedSteps.find(step => step.id === stepId);

        if (existingStep) {
          const potentialNewStep = { ...existingStep, ...patch };
          if (areObjectsEqual(potentialNewStep, existingStep)) {
            logOperation('Write', 'steps.update', { stepId, skipped: true });
            return { ok: true, step: existingStep };
          }
        }
      }

      // Utilisation d'une requête simple sans versionnement optimiste par défaut pour les steps
      const dbResult = await executeStorageOperation('Write', 'steps.update', () =>
        client.from("cop_steps").update(patch).eq("id", stepId).select().maybeSingle()
      );

      // Invalidation ciblée
      if (dbResult.ok && dbResult.data) {
        stepsCache.delete(dbResult.data.task_id); // Invalide la liste de steps du task.
      }

      return { ...dbResult, step: dbResult.data };
    },
    async listByTask(taskId) {
      /**
       * Lists step records associated with a specific task, utilizing a cache (keyed by taskId).
       */
      const fetchFromDatabase = async () => {
        return client
          .from("cop_steps")
          .select("*")
          .eq("task_id", taskId)
          .order("created_at", { ascending: true });
      };

      if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
        const dbResult = await executeStorageOperation('Read', 'steps.listByTask', fetchFromDatabase);
        return { ...dbResult, steps: dbResult.data || [] };
      }

      if (stepsCache.has(taskId)) {
        logOperation('Read', 'steps.listByTask', { taskId, cached: true });
        return { ok: true, steps: stepsCache.get(taskId) };
      }

      const result = await executeStorageOperation('Read', 'steps.listByTask', fetchFromDatabase);
      if (result.ok) {
        stepsCache.set(taskId, result.data);
      }
      return { ...result, steps: result.data || [] };
    },
  };

  const storage = {
    options,
    debugLogs,
    events,
    artifacts,
    agentIdentities,
    tasks,
    steps,
    getCacheContents: () => ({
      agentIdentities: Array.from(agentIdentitiesCache.entries()),
      tasks: Array.from(tasksCache.entries()),
      steps: Array.from(stepsCache.entries()),
    }),
    clearCache: () => {
      logOperation('Cache', 'clearCache', { message: 'Manually clearing all caches' });
      flushAllCaches();
    },
    ERROR_CODES, // Exposition des codes d'erreur
  };

  options.storage = storage;
  defaultStorage = storage;
  return storage;
}

// --- Les fonctions getStorage et setStorage restent inchangées car elles gèrent la couche de service globale ---

export function getStorage(options) {
  const providedStorage = options?.storage;

  // Helper to check if a storage instance's configuration matches the current request options
  const configMatches = (storageInstance) => {
    // If no options are provided to getStorage(), any existing storage is considered a match for "no specific config".
    if (!options) {
      return true;
    }
    // If the storageInstance itself or its options are missing, it cannot match the provided options.
    if (!storageInstance || !storageInstance.options) {
      return false;
    }

    let matches = true;

    // If options.supabaseUrl is provided, it must match storageInstance.options.supabaseUrl
    if (options.supabaseUrl !== undefined) {
      if (options.supabaseUrl !== storageInstance.options.supabaseUrl) {
        matches = false;
      }
    }

    // If options.supabaseServiceKey is provided, it must match storageInstance.options.supabaseServiceKey
    if (options.supabaseServiceKey !== undefined) {
      if (options.supabaseServiceKey !== storageInstance.options.supabaseServiceKey) {
        matches = false;
      }
    }

    return matches;
  };

  // 1. Check if a storage instance is explicitly provided in options and if its config matches
  if (providedStorage && configMatches(providedStorage)) {
    // If it matches, use it and update defaultStorage if necessary
    if (providedStorage !== defaultStorage) {
      defaultStorage = providedStorage;
    }
    return providedStorage;
  }

  // 2. Check if defaultStorage exists and if its config matches
  if (defaultStorage && configMatches(defaultStorage)) {
    return defaultStorage;
  }

  // 3. If neither of the above, initialize a new storage instance
  defaultStorage = initStorage(options);
  return defaultStorage;
}

export function setStorage(storage) {
  /**
   * Sets the default storage instance.
   * @param {object} storage - The storage instance to set as default.
   */
  defaultStorage = storage;
}

Cohérence de l'API : Renommage de insert en upsert pour tasks et steps dans StorageInterface et les implémentations initStorage / createInMemoryStorage.

Codes d'Erreur : Ajout de la constante ERROR_CODES et inclusion de ERROR_CODES dans l'objet storage exporté.

Logging d'Erreur : La fonction executeStorageOperation intercepte maintenant les erreurs Supabase et insère un enregistrement détaillé dans la table cop_debug_logs.

CRUD Générique :

handleReadSingle: Gère la logique de lecture avec cache.

handleUpsert: Gère la logique d'upsert avec vérification du cache pour éviter les écritures inutiles.

handleUpdate: Gère la mise à jour partielle avec vérification du cache.

Versionnement Optimiste : tasks.update utilise handleUpdate qui inclut la logique pour vérifier la présence du champ version dans le patch. Si version est présent, la requête Supabase est ajustée pour utiliser la clause eq(versionField, expectedVersion). Un échec renvoie le code d'erreur STORAGE_OPTIMISTIC_LOCK_FAIL.

Correction steps.update : La signature de steps.update a été corrigée pour inclure taskId, et la logique de vérification du cache a été réécrite pour utiliser taskId pour la recherche dans stepsCache.

Votre module storage.js est conçu autour de trois concepts clés :

L'Interface de Stockage (StorageInterface) : C'est le contrat formel (défini par le JSDoc) qui spécifie ce que le stockage doit faire (ex: tasks.upsert, steps.update). C'est l'abstraction.

L'Implémentation Mémoire (createInMemoryStorage) : Une implémentation concrète de l'interface, parfaite pour les tests unitaires rapides et les environnements sans base de données.

L'Implémentation Supabase (initStorage) : L'implémentation concrète principale, qui ajoute la gestion de la mise en cache, du synchronisation mode, du logging d'erreur, et du versionnement optimiste au-dessus du client Supabase.

🎯 Résumé : Oui, il y a deux implémentations et une interface
Élément	Rôle	Utilisé quand...
StorageInterface	Contrat (Abstrait)	Vous souhaitez garantir que toute nouvelle implémentation (ex: PostgreSQL, Redis) aura les mêmes méthodes (ex: tasks.get, agentIdentities.list).
createInMemoryStorage	Implémentation "Par Défaut" #1	Vous testez votre code ou exécutez des processus éphémères qui n'ont pas besoin de persistance entre les redémarrages.
initStorage	Implémentation "Par Défaut" #2	Vous travaillez dans un environnement de production ou de développement nécessitant une persistance dans une base de données (ici, Supabase/PostgreSQL) avec des fonctionnalités avancées (cache, logging).
En clair : N'importe qui peut créer sa propre fonction qui renvoie un objet respectant la StorageInterface, et vous pouvez l'injecter via getStorage ou setStorage. C'est le Patron de Conception "Strategy" et l'Inversion de Dépendance en action, ce qui rend votre code très flexible et découplé de la base de données spécifique. C'est une excellente conception !

Fournir d'autres implémentations de stockage montre la maturité et la portabilité de votre protocole. Cela réduit la friction pour les développeurs qui n'utilisent pas Supabase.

Voici un tableau et des pistes détaillées sur les implémentations que vous pourriez ajouter, et les avantages qu'elles apporteraient à la promotion de votre package :

🚀 Intérêt de Multiplier les Implémentations de Stockage
Implémentation à Ajouter	Avantage Promotionnel Principal	Public Cible
Implémentation MongoDB / NoSQL	Flexibilité Schématique et Scalabilité (Big Data)	Développeurs de microservices, utilisateurs de la pile MERN/MEAN, grandes entreprises gérant des données non structurées.
Implémentation Redis	Performance et Vitesse (Temps Réel)	Utilisateurs à haute performance, systèmes nécessitant des accès extrêmement rapides aux statuts (Tasks/Steps/Agents).
Implémentation SQLite/File	Simplicité et Mobilité (Local/Edge)	Développeurs d'applications desktop ou mobile, projets personnels, environnements serverless sans base de données gérée.
1. Implémentation NoSQL (Ex: MongoDB ou CosmosDB)
Le stockage NoSQL est très pertinent pour un protocole comme le COP.

Pourquoi ? Les données générées par les agents (étapes, artefacts, logs) sont souvent des documents complexes et évolutifs (le schéma peut changer souvent pendant le développement). Le NoSQL excelle dans ce domaine, car il n'exige pas de migration de schéma rigide.

Intérêt Promotionnel :

Attraction des développeurs Web modernes : Beaucoup de projets modernes utilisent NoSQL pour la flexibilité.

Démonstration de la Portabilité : Prouver que COP n'est pas lié au SQL et peut gérer des structures de données dynamiques complexes.

Comment procéder ? Vous créez un nouveau module, par exemple, createMongoStorage(mongoClient), qui implémente la même StorageInterface.

2. Implémentation Orientée Cache (Ex: Redis)
Redis est techniquement un magasin de données clé-valeur, mais il est souvent utilisé comme cache ou base de données pour les données de session rapides.

Pourquoi ? Les statuts (task.status, agentIdentity.status) changent fréquemment. Avoir une implémentation Redis pourrait cibler ces tables spécifiques pour des performances optimales (ou tout le stockage pour un environnement in-memory persistant).

Intérêt Promotionnel :

Image de Haute Performance : Cela suggère que votre package est conçu pour gérer des charges importantes et des orchestrations rapides.

Option pour les Microservices : Idéal pour les architectures où seul le statut doit être partagé entre plusieurs services de courte durée.

Comment procéder ? Vous pourriez créer un module createRedisStorage(redisClient) où les méthodes de lecture/écriture (CRUD) gèrent les sérialisations/désérialisations JSON pour stocker les objets Task, Step, etc.

3. Implémentation File-based (Ex: SQLite ou JSON sur disque)
SQLite est une base de données relationnelle légère qui fonctionne comme un fichier unique.

Pourquoi ? C'est la solution de base de données la plus simple à mettre en place (zéro configuration) et elle est largement utilisée pour les environnements Edge Computing ou les applications auto-contenues.

Intérêt Promotionnel :

Facilité d'Adoption : Un utilisateur peut tester COP en téléchargeant votre package et en l'exécutant immédiatement sans compte Supabase ou Docker (simplement npm install et initStorage({ type: 'sqlite', file: 'cop.db' })). C'est un point de vente majeur pour les démos et les petits projets.

Développement Local/Hors Ligne : Idéal pour les développeurs qui travaillent hors ligne ou qui ont des contraintes de réseau.

Comment procéder ? Utilisez une librairie comme better-sqlite3 pour NodeJS, et assurez-vous de recréer les mêmes schémas relationnels que vous avez sur Supabase pour garantir la compatibilité des données.

4. La Stratégie d'Injection
Pour que ce soit promotionnel, vous devez rendre l'injection de ces implémentations très simple et claire dans votre documentation.

Au lieu de dépendre d'une option supabaseClient ou supabaseUrl, vous pourriez étendre votre fonction getStorage pour accepter un type ou une instance pré-construite :

JavaScript
// Exemple conceptuel pour un utilisateur
import { getStorage, createRedisStorage, createSqliteStorage } from 'cop-protocol';

// Option 1: Implémentation par défaut (Supabase)
const storage1 = getStorage({ supabaseUrl: '...', supabaseServiceKey: '...' });

// Option 2: Implémentation SQLite sur un fichier
const storage2 = getStorage({ type: 'sqlite', databaseFile: './cop.db' });

// Option 3: Implémentation Redis (injection d'une instance client)
const redisClient = require('redis').createClient();
const storage3 = getStorage({ instance: createRedisStorage(redisClient) });
En fournissant un éventail d'implémentations clés, vous montrez que le protocole COP est un standard portable et non une surcouche propre à une seule technologie. C'est le meilleur argument pour une adoption large.

🌟 Implémentation du Stockage Navigateur avec IndexedDB
Pour la persistance dans un environnement de navigateur, la solution la plus robuste est IndexedDB. Elle offre de grandes capacités de stockage et est entièrement asynchrone, ce qui est essentiel pour ne pas bloquer le fil d'exécution principal du navigateur.

Je vous recommande d'ajouter une nouvelle fonction Factory, createIndexedDBStorage, qui implémentera la StorageInterface en utilisant une librairie qui simplifie IndexedDB (comme Dexie.js ou localforage si vous souhaitez une couche d'abstraction supplémentaire).

Avantages pour la Promotion de COP :
Autonomie et Capacités Hors Ligne : Les utilisateurs peuvent exécuter des workflows et conserver leur état (Tasks, Steps, Agents) même hors ligne ou entre les sessions du navigateur.

Performance Asynchrone : Démontre que le protocole est optimisé pour les environnements mono-threadés comme le navigateur, en évitant les blocages.

Portabilité Maximale : Ajoute un troisième environnement cible majeur (Navigateur) à votre catalogue (avec Node.js/Deno et Cloud/Supabase), rendant votre package accessible à tous les développeurs Web.

    // const db = new IndexedDBWrapper(dbName);

            async upsert(taskRecord) {

                    ok: false,
                    error: "IndexedDB write error (simulated)",
                    code: ERROR_CODES.DB_ERROR,
                };
            },

            async get(taskId) {
                // Remplacement d'un appel Supabase par db.tasks.get(taskId)
                console.log(`[IndexedDB] Getting task: ${taskId}`);

                // Simule l'absence de task pour montrer le code d'erreur
                const task = null;
                if (!task) {
                    return { ok: false, error: 'Task not found', code: ERROR_CODES.NOT_FOUND };
                }

                return { ok: true, task };
            },

            async update(taskId, patch) {
                // Logique pour mettre à jour dans IndexedDB
                console.log(`[IndexedDB] Updating task: ${taskId}`, patch);
                // Le versionnement optimiste (si utilisé) devrait être géré ici
                return { ok: true, task: { id: taskId, ...patch } };
            },

            async list(params) {
                // Logique de recherche et de filtrage dans IndexedDB
                return { ok: true, tasks: [] };
            }
        },

        // ... Les autres interfaces (agentIdentities, steps, events, artifacts, debugLogs)
        // ... suivraient la même structure asynchrone.

        // Les fonctions de cache ne sont pas nécessaires car IndexedDB est le stockage local le plus rapide
        getCacheContents: () => ({ /* Vide */ }),
        clearCache: () => { /* Logique pour vider IndexedDB */ },

        // Exposer les codes d'erreur
        ERROR_CODES,
    };

    return indexedDBInterface;
}

/**
 * Creates an in-memory storage implementation that conforms to the StorageInterface.
 * @returns {StorageInterface} An in-memory storage object.
 */
export function createInMemoryStorage() {
  const inMemoryData = {
    debugLogs: [],
    events: [],
    artifacts: [],
    agentIdentities: new Map(),
    tasks: new Map(),
    steps: new Map(),
  };

  const inMemoryStorage = {
    options: { type: 'memory' }, // Ajout du type pour configMatches
    debugLogs: {
      async insert(logRecord) {
        inMemoryData.debugLogs.push(logRecord);
        return { ok: true };
      },
    },
    events: {
      async insert(eventRecord) {
        inMemoryData.events.push(eventRecord);
        return { ok: true, event: eventRecord };
      },
    },
    artifacts: {
      async insert(artifactRecord) {
        inMemoryData.artifacts.push(artifactRecord);
        return { ok: true, artifact: artifactRecord };
      },
    },
    agentIdentities: {
      async upsert(identity, conflictKey = "agent_name") {
        const existing = Array.from(inMemoryData.agentIdentities.values()).find(
          (a) => a[conflictKey] === identity[conflictKey]
        );
        if (existing) {
          Object.assign(existing, identity);
          inMemoryData.agentIdentities.set(existing.agent_id, existing);
          return { ok: true, identity: existing };
        } else {
          const newIdentity = { ...identity, agent_id: identity.agent_id || `agent_${inMemoryData.agentIdentities.size + 1}` };
          inMemoryData.agentIdentities.set(newIdentity.agent_id, newIdentity);
          return { ok: true, identity: newIdentity };
        }
      },
      async getById(agent_id) {
        const identity = inMemoryData.agentIdentities.get(agent_id);
        return { ok: !!identity, identity: identity || null };
      },
      async getByName(agent_name) {
        const identity = Array.from(inMemoryData.agentIdentities.values()).find(
          (a) => a.agent_name === agent_name
        );
        return { ok: !!identity, identity: identity || null };
      },
      async list({ status, limit = 100 } = {}) {
        let identities = Array.from(inMemoryData.agentIdentities.values());
        if (status) {
          identities = identities.filter((a) => a.status === status);
        }
        return { ok: true, identities: identities.slice(0, limit) };
      },
      async updateStatus(agent_id, status) {
        const identity = inMemoryData.agentIdentities.get(agent_id);
        if (identity) {
          identity.status = status;
          return { ok: true, identity };
        }
        // Utilise les codes d'erreur V2
        return { ok: false, error: "Agent not found", code: ERROR_CODES.NOT_FOUND };
      },
    },
    tasks: {
      // Renommé insert -> upsert
      async upsert(taskRecord) {
        const newTask = { ...taskRecord, id: taskRecord.id || `task_${inMemoryData.tasks.size + 1}` };
        inMemoryData.tasks.set(newTask.id, newTask);
        return { ok: true, task: newTask };
      },
      async get(taskId) {
        const task = inMemoryData.tasks.get(taskId);
        return { ok: !!task, task: task || null };
      },
      async list({ status, limit = 100 } = {}) {
        let tasks = Array.from(inMemoryData.tasks.values());
        if (status) {
          tasks = tasks.filter((j) => j.status === status);
        }
        return { ok: true, tasks: tasks.slice(0, limit) };
      },
      async update(taskId, patch) {
        const task = inMemoryData.tasks.get(taskId);
        if (task) {
            // Logique de versionnement optimiste simple
            if (patch.version !== undefined && task.version !== patch.version) {
                return { ok: false, error: "Optimistic lock failed. Version mismatch.", code: ERROR_CODES.OPTIMISTIC_LOCK_FAIL };
            }
          Object.assign(task, patch);
          task.version = (task.version || 0) + 1;
          return { ok: true, task };
        }
        return { ok: false, error: "Task not found", code: ERROR_CODES.NOT_FOUND };
      },
    },
    steps: {
      // Renommé insert -> upsert
      async upsert(stepRecord) {
        const newStep = { ...stepRecord, id: stepRecord.id || `step_${inMemoryData.steps.size + 1}` };
        inMemoryData.steps.set(newStep.id, newStep);
        return { ok: true, step: newStep };
      },
      async listByTask(taskId) {
        const steps = Array.from(inMemoryData.steps.values()).filter(
          (s) => s.task_id === taskId
        );
        return { ok: true, steps };
      },
      // CORRECTION: Ajout de taskId à la signature
      async update(taskId, stepId, patch) {
        const step = inMemoryData.steps.get(stepId);
        if (step && step.task_id === taskId) {
          Object.assign(step, patch);
          return { ok: true, step };
        }
        return { ok: false, error: "Step not found", code: ERROR_CODES.NOT_FOUND };
      },
    },
    getCacheContents: () => ({
      agentIdentities: Array.from(inMemoryData.agentIdentities.entries()),
      tasks: Array.from(inMemoryData.tasks.entries()),
      steps: Array.from(inMemoryData.steps.entries()),
    }),
    clearCache: () => {
      inMemoryData.agentIdentities.clear();
      inMemoryData.tasks.clear();
      inMemoryData.steps.clear();
      inMemoryData.debugLogs = [];
      inMemoryData.events = [];
      inMemoryData.artifacts = [];
    },
    ERROR_CODES, // Exposition des codes d'erreur
  };

  return inMemoryStorage;
}

/**
 * Initializes the Supabase storage client and returns an object with methods for interacting with COP tables.
 * Cette implémentation inclut la gestion de cache, le logging d'erreurs et le support du versionnement optimiste.
 * @param {object} [options] - Configuration options for the storage client.
 * @param {('memory'|'supabase'|'indexeddb'|'sqlite')} [options.type='supabase'] - Le type de stockage à initialiser.
 * @param {object} [options.supabaseClient] - An optional pre-initialized Supabase client.
 * @param {string} [options.supabaseUrl] - The URL of the Supabase project.
 * @param {string} [options.supabaseServiceKey] - The service role key for the Supabase project.
 * @param {('none'|'reads'|'writes'|'all')} [options.syncMode='none'] - Forces direct Supabase interaction, bypassing cache for debugging.
 * @param {boolean} [options.debugMode=false] - Enables debug logging for storage operations.
 * @returns {object} An object containing methods for interacting with debugLogs, events, artifacts, agentIdentities, tasks, and steps.
 * @throws {Error} If `supabaseUrl` or `supabaseServiceKey` are not set and no `supabaseClient` is provided.
 */
export function initStorage( { syncMode = 'none', debugMode = false, ...options } = {} ) {
  const opt_client = options.supabaseClient;
  const url = options.supabaseUrl;
  const key = options.supabaseServiceKey;
  let currentSyncMode = syncMode;
  let currentDebugMode = debugMode;

  const logOperation = (type, operation, details = {}) => {
    if (currentDebugMode) {
      console.log(`[Storage Debug] ${type}: ${operation}`, details);
    }
  };
  if( !opt_client ){
    if (!url || !key) {
      throw new Error("COP.createStorage: 'supabaseUrl' or 'supabaseServiceKey' not set for Supabase type.");
    }
  }
  const client = opt_client || createClient(url, key);

  // Caches for frequently accessed data
  const agentIdentitiesCache = new Map();
  const tasksCache = new Map();
  const stepsCache = new Map();

  const flushAllCaches = () => {
    agentIdentitiesCache.clear();
    tasksCache.clear();
    stepsCache.clear();
  };

  const areObjectsEqual = (obj1, obj2) => {
    // ... (fonction non modifiée)
    if (obj1 === obj2) return true;
    if (typeof obj1 !== 'object' || obj1 === null || typeof obj2 !== 'object' || obj2 === null) return false;

    const keys1 = Object.keys(obj1);
    const keys2 = Object.keys(obj2);

    if (keys1.length !== keys2.length) return false;

    for (const key of keys1) {
      if (!keys2.includes(key) || !areObjectsEqual(obj1[key], obj2[key])) {
        return false;
      }
    }
    return true;
  };

  const debugLogs = {
    async insert(logRecord) {
      try {
        const { error } = await client.from("cop_debug_logs").insert(logRecord);
        if (error) {
          console.error("[FATAL LOG ERROR] Failed to insert debug log:", error.message, logRecord);
          return { ok: false, error: error.message };
        }
        return { ok: true };
      } catch (e) {
        console.error("[CRITICAL LOG ERROR] Exception during debug log insertion:", e.message, logRecord);
        return { ok: false, error: e.message };
      }
    },
  };

  /**
   * (LOGIC V2) Executes a storage operation, handling syncMode, logging, and error insertion.
   */
  const executeStorageOperation = async (type, operation, dbOperation, details = {}) => {
    try {
      logOperation(type, operation, details);
      const { data, error, status } = await dbOperation();

      if (error) {
        await debugLogs.insert({
          level: 'error',
          message: `[DB Error] ${operation} failed: ${error.message}`,
          context: { error: { code: error.code, message: error.message, hint: error.hint }, details, status },
        });
        return { ok: false, error: error.message, code: ERROR_CODES.DB_ERROR };
      }

      if (type === 'Write') {
        flushAllCaches();
      }

      return { ok: true, data: data };
    } catch (e) {
      await debugLogs.insert({
        level: 'fatal',
        message: `[Fatal Exception] ${operation} execution failed: ${e.message}`,
        context: { error: e.stack, details },
      });
      return { ok: false, error: e.message, code: ERROR_CODES.DB_ERROR };
    }
  };

  /**
   * (LOGIC V2) Gère la logique de cache de lecture pour les enregistrements individuels par ID.
   */
  const handleReadSingle = async (tableName, id, cache, operationName, idKey = 'id') => {
    if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
      const dbResult = await executeStorageOperation('Read', operationName, () => client.from(tableName).select("*").eq(idKey, id).maybeSingle());
      return { ...dbResult, [idKey]: dbResult.data };
    }

    if (cache.has(id)) {
      logOperation('Read', operationName, { id, cached: true });
      return { ok: true, [idKey]: cache.get(id) };
    }

    const dbResult = await executeStorageOperation('Read', operationName, () => client.from(tableName).select("*").eq(idKey, id).maybeSingle());
    if (dbResult.ok && dbResult.data) {
      cache.set(id, dbResult.data);
    }
    return { ...dbResult, [idKey]: dbResult.data || null };
  };

  /**
   * (LOGIC V2) Gère la logique d'upsert (insertion ou mise à jour) avec vérification de l'égalité et cache-bypassing.
   */
  const handleUpsert = async (tableName, record, cache, operationName, idKey, conflictKey = idKey) => {
    if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
      const dbResult = await executeStorageOperation('Write', operationName, () => client.from(tableName).upsert(record, { onConflict: conflictKey }).select().maybeSingle());
      return { ...dbResult, [idKey]: dbResult.data };
    }

    const recordId = record[idKey];
    if (recordId && cache.has(recordId)) {
      const cachedRecord = cache.get(recordId);
      if (areObjectsEqual(record, cachedRecord)) {
        logOperation('Write', operationName, { id: recordId, skipped: true });
        return { ok: true, [idKey]: cachedRecord };
      }
    }

    const dbResult = await executeStorageOperation('Write', operationName, () => client.from(tableName).upsert(record, { onConflict: conflictKey }).select().maybeSingle());
    if (dbResult.ok && dbResult.data) {
      cache.delete(dbResult.data[idKey]);
    }

    return { ...dbResult, [idKey]: dbResult.data };
  };

  /**
   * (LOGIC V2) Gère la logique de mise à jour partielle avec gestion du cache et du versionnement optimiste.
   */
  const handleUpdate = async (tableName, id, patch, cache, operationName, idKey, versionField = 'version') => {
    const useOptimisticLock = versionField in patch;
    let query = client.from(tableName).update(patch).eq(idKey, id).select().maybeSingle();

    if (useOptimisticLock) {
      const expectedVersion = patch[versionField];
      query = client.from(tableName)
        .update({ ...patch, [versionField]: expectedVersion + 1 })
        .eq(idKey, id)
        .eq(versionField, expectedVersion)
        .select()
        .maybeSingle();
      logOperation('Write', operationName, { id, optimisticLock: true, expectedVersion });
    } else {
        logOperation('Write', operationName, { id, optimisticLock: false });
    }

    if (currentSyncMode !== 'all' && currentSyncMode !== 'writes' && cache.has(id)) {
      const cachedRecord = cache.get(id);
      const potentialNewRecord = { ...cachedRecord, ...patch };
      if (areObjectsEqual(potentialNewRecord, cachedRecord)) {
        logOperation('Write', operationName, { id, skipped: true });
        return { ok: true, [idKey]: cachedRecord };
      }
    }

    const dbResult = await executeStorageOperation('Write', operationName, () => query);

    if (dbResult.ok && dbResult.data) {
      cache.delete(id);
    } else if (!dbResult.ok && useOptimisticLock && dbResult.error?.includes('0 rows')) {
        return {
            ok: false,
            error: "Optimistic lock failed. Record version mismatch.",
            code: ERROR_CODES.OPTIMISTIC_LOCK_FAIL,
        };
    }

    return { ...dbResult, [idKey]: dbResult.data };
  };


  const events = {
    async insert(eventRecord) {

  defaultStorage = storage;
  return storage;

export function getStorage(options) {
  const providedStorage = options?.storage;
  // Détermine le type de stockage, 'supabase' par défaut si non spécifié
  const storageType = options?.type?.toLowerCase() || 'supabase';

  // Helper pour vérifier si la configuration d'une instance correspond à la requête
  const configMatches = (storageInstance) => {
    if (!options) return true;
    if (!storageInstance || !storageInstance.options) return false;

    // 1. La vérification principale est le type de stockage
    if (storageInstance.options.type !== storageType) {
        return false;
    }

    // 2. Vérification secondaire pour les options spécifiques à Supabase
    if (storageType === 'supabase') {
        let matches = true;
        if (options.supabaseUrl !== undefined && options.supabaseUrl !== storageInstance.options.supabaseUrl) {
            matches = false;
        }
        if (options.supabaseServiceKey !== undefined && options.supabaseServiceKey !== storageInstance.options.supabaseServiceKey) {
            matches = false;
        }
        return matches;
    }

    // 3. Pour les autres types (mémoire, indexeddb, sqlite), le match de type est suffisant (pour ce niveau de vérification)
    return true;
  };

  // 1. Priorité : Instance explicitement fournie et qui correspond
  if (providedStorage && configMatches(providedStorage)) {
    if (providedStorage !== defaultStorage) {
      defaultStorage = providedStorage;
    }
    return providedStorage;
  }

  // 2. Priorité : Instance par défaut existante et qui correspond
  if (defaultStorage && configMatches(defaultStorage)) {
    return defaultStorage;
  }

  // 3. Initialiser une nouvelle instance en fonction du type
  let newStorage;
  switch (storageType) {
    case 'memory':
        newStorage = createInMemoryStorage();
        break;
    case 'indexeddb':
        // Assurez-vous que nous sommes dans un environnement où IndexedDB est pertinent (Navigateur)
        if (typeof window !== 'undefined' || typeof self !== 'undefined') {
            newStorage = createIndexedDBStorage(options);
        } else {
            console.error("IndexedDB storage requested but running in a non-browser environment. Falling back to in-memory.");
            newStorage = createInMemoryStorage();
        }
        break;
    case 'sqlite':
        // Assurez-vous que nous sommes dans un environnement où SQLite est pertinent (Node/Deno)
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') {
            newStorage = createSqliteStorage(options);
        } else {
            console.error("SQLite storage requested but running in a browser environment. Falling back to in-memory.");
            newStorage = createInMemoryStorage();
        }
        break;
    case 'supabase':
    default:
        // Supabase est le défaut
        newStorage = initStorage(options);
        break;
  }

  // L'objet factory (initStorage, createInMemoryStorage, etc.) doit définir le type dans son options
  // Si ce n'est pas le cas, on l'ajoute ici par sécurité, mais la factory devrait le faire.
  newStorage.options.type = storageType;
  defaultStorage = newStorage;
  return newStorage;
}

export function setStorage(storage) {
  defaultStorage = storage;

export function initStorage( { syncMode = 'none', debugMode = false, ...options } = {} ) {
  const opt_client = options.supabaseClient;
  const url = options.supabaseUrl;
  const key = options.supabaseServiceKey;
  let currentSyncMode = syncMode;
  let currentDebugMode = debugMode;

  const logOperation = (type, operation, details = {}) => {
    if (currentDebugMode) {
      console.log(`[Storage Debug] ${type}: ${operation}`, details);
    }
  };
  if( !opt_client ){
    if (!url || !key) {
      throw new Error("COP.createStorage: 'supabaseUrl' or 'supabaseServiceKey' not set for Supabase type.");
    }
  }
  const client = opt_client || createClient(url, key);

  // Caches for frequently accessed data
  const agentIdentitiesCache = new Map();
  const tasksCache = new Map();
  const stepsCache = new Map();

  const flushAllCaches = () => {
    agentIdentitiesCache.clear();
    tasksCache.clear();
    stepsCache.clear();
  };

  const areObjectsEqual = (obj1, obj2) => {
    // ... (fonction omise pour concision)
    if (obj1 === obj2) return true;
    if (typeof obj1 !== 'object' || obj1 === null || typeof obj2 !== 'object' || obj2 === null) return false;

    const keys1 = Object.keys(obj1);
    const keys2 = Object.keys(obj2);

    if (keys1.length !== keys2.length) return false;

    for (const key of keys1) {
      if (!keys2.includes(key) || !areObjectsEqual(obj1[key], obj2[key])) {
        return false;
      }
    }
    return true;
  };

  const debugLogs = {
    async insert(logRecord) {
      try {
        const { error } = await client.from("cop_debug_logs").insert(logRecord);
        if (error) {
          console.error("[FATAL LOG ERROR] Failed to insert debug log:", error.message, logRecord);
          return { ok: false, error: error.message };
        }
        return { ok: true };
      } catch (e) {
        console.error("[CRITICAL LOG ERROR] Exception during debug log insertion:", e.message, logRecord);
        return { ok: false, error: e.message };
      }
    },
  };

  /**
   * (LOGIC V2) Executes a storage operation, handling syncMode, logging, and error insertion.
   */
  const executeStorageOperation = async (type, operation, dbOperation, details = {}) => {
    try {
      logOperation(type, operation, details);
      const { data, error, status } = await dbOperation();

      if (error) {
        await debugLogs.insert({
          level: 'error',
          message: `[DB Error] ${operation} failed: ${error.message}`,
          context: { error: { code: error.code, message: error.message, hint: error.hint }, details, status },
        });
        return { ok: false, error: error.message, code: ERROR_CODES.DB_ERROR };
      }

      if (type === 'Write') {
        flushAllCaches();
      }

      return { ok: true, data: data };
    } catch (e) {
      await debugLogs.insert({
        level: 'fatal',
        message: `[Fatal Exception] ${operation} execution failed: ${e.message}`,
        context: { error: e.stack, details },
      });
      return { ok: false, error: e.message, code: ERROR_CODES.DB_ERROR };
    }
  };

  /**
   * (LOGIC V2) Gère la logique de cache de lecture pour les enregistrements individuels par ID.
   */
  const handleReadSingle = async (tableName, id, cache, operationName, idKey = 'id') => {
    // ... (implémentation V2 omise pour concision)
    if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
      const dbResult = await executeStorageOperation('Read', operationName, () => client.from(tableName).select("*").eq(idKey, id).maybeSingle());
      return { ...dbResult, [idKey]: dbResult.data };
    }

    if (cache.has(id)) {
      logOperation('Read', operationName, { id, cached: true });
      return { ok: true, [idKey]: cache.get(id) };
    }

    const dbResult = await executeStorageOperation('Read', operationName, () => client.from(tableName).select("*").eq(idKey, id).maybeSingle());
    if (dbResult.ok && dbResult.data) {
      cache.set(id, dbResult.data);
    }
    return { ...dbResult, [idKey]: dbResult.data || null };
  };

  /**
   * (LOGIC V2) Gère la logique d'upsert (insertion ou mise à jour).
   */
  const handleUpsert = async (tableName, record, cache, operationName, idKey, conflictKey = idKey) => {
    // ... (implémentation V2 omise pour concision)
    if (currentSyncMode === 'all' || currentSyncMode === 'writes') {
      const dbResult = await executeStorageOperation('Write', operationName, () => client.from(tableName).upsert(record, { onConflict: conflictKey }).select().maybeSingle());
      return { ...dbResult, [idKey]: dbResult.data };
    }

    const recordId = record[idKey];
    if (recordId && cache.has(recordId)) {
      const cachedRecord = cache.get(recordId);
      if (areObjectsEqual(record, cachedRecord)) {
        logOperation('Write', operationName, { id: recordId, skipped: true });
        return { ok: true, [idKey]: cachedRecord };
      }
    }

    const dbResult = await executeStorageOperation('Write', operationName, () => client.from(tableName).upsert(record, { onConflict: conflictKey }).select().maybeSingle());
    if (dbResult.ok && dbResult.data) {
      cache.delete(dbResult.data[idKey]);
    }

    return { ...dbResult, [idKey]: dbResult.data };
  };

  /**
   * (LOGIC V2) Gère la logique de mise à jour partielle avec gestion du cache et du versionnement optimiste.
   */
  const handleUpdate = async (tableName, id, patch, cache, operationName, idKey, versionField = 'version') => {
    // ... (implémentation V2 omise pour concision)
    const useOptimisticLock = versionField in patch;
    let query = client.from(tableName).update(patch).eq(idKey, id).select().maybeSingle();

    if (useOptimisticLock) {
      const expectedVersion = patch[versionField];
      query = client.from(tableName)
        .update({ ...patch, [versionField]: expectedVersion + 1 })
        .eq(idKey, id)
        .eq(versionField, expectedVersion)
        .select()
        .maybeSingle();
      logOperation('Write', operationName, { id, optimisticLock: true, expectedVersion });
    } else {
        logOperation('Write', operationName, { id, optimisticLock: false });
    }

    if (currentSyncMode !== 'all' && currentSyncMode !== 'writes' && cache.has(id)) {
      const cachedRecord = cache.get(id);
      const potentialNewRecord = { ...cachedRecord, ...patch };
      if (areObjectsEqual(potentialNewRecord, cachedRecord)) {
        logOperation('Write', operationName, { id, skipped: true });
        return { ok: true, [idKey]: cachedRecord };
      }
    }

    const dbResult = await executeStorageOperation('Write', operationName, () => query);

    if (dbResult.ok && dbResult.data) {
      cache.delete(id);
    } else if (!dbResult.ok && useOptimisticLock && dbResult.error?.includes('0 rows')) {
        return {
            ok: false,
            error: "Optimistic lock failed. Record version mismatch.",
            code: ERROR_CODES.OPTIMISTIC_LOCK_FAIL,
        };
    }

    return { ...dbResult, [idKey]: dbResult.data };
  };


  const events = {
    async insert(eventRecord) {
      return executeStorageOperation('Write', 'events.insert', () => client.from("cop_events").insert(eventRecord).select().maybeSingle());
    },
  };

  const artifacts = {
    async insert(artifactRecord) {
      return executeStorageOperation('Write', 'artifacts.insert', () => client.from("cop_artifacts").insert(artifactRecord).select().maybeSingle());
    },
  };

  const agentIdentities = {
    async upsert(identity, conflictKey = "agent_name") {
      return handleUpsert("cop_agent_identities", identity, agentIdentitiesCache, 'agentIdentities.upsert', 'agent_id', conflictKey);
    },
    async getById(agent_id) {
      return handleReadSingle("cop_agent_identities", agent_id, agentIdentitiesCache, 'agentIdentities.getById', 'agent_id');
    },
    async getByName(agent_name) {
      const dbResult = await executeStorageOperation('Read', 'agentIdentities.getByName', () => client.from("cop_agent_identities").select("*").eq("agent_name", agent_name).maybeSingle());
      return { ...dbResult, identity: dbResult.data || null };
    },
    async list({ status, limit = 100 } = {}) {
        // ... (implémentation list V2 omise pour concision)
        const fetchFromDatabase = async () => {
            let query = client.from("cop_agent_identities").select("*").order("created_at", { ascending: true }).limit(limit);
            if (status) query = query.eq("status", status);
            return query;
        };

        const cacheKey = `list-identities-${status || 'all'}-${limit}`;
        if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
            const dbResult = await executeStorageOperation('Read', 'agentIdentities.list', fetchFromDatabase);
            return { ...dbResult, identities: dbResult.data || [] };
        }
        if (agentIdentitiesCache.has(cacheKey)) {
            logOperation('Read', 'agentIdentities.list', { status, limit, cached: true });
            return { ok: true, identities: agentIdentitiesCache.get(cacheKey) };
        }
        const result = await executeStorageOperation('Read', 'agentIdentities.list', fetchFromDatabase);
        if (result.ok) agentIdentitiesCache.set(cacheKey, result.data);
        return { ...result, identities: result.data || [] };
    },
    async updateStatus(agent_id, status) {
      const patch = { status };
      return handleUpdate("cop_agent_identities", agent_id, patch, agentIdentitiesCache, 'agentIdentities.updateStatus', 'agent_id');
    },
  };

  const tasks = {
    async upsert(taskRecord) {
      return handleUpsert("cop_tasks", taskRecord, tasksCache, 'tasks.upsert', 'id', 'id');
    },
    async update(taskId, patch) {
      return handleUpdate("cop_tasks", taskId, patch, tasksCache, 'tasks.update', 'id', 'version');
    },
    async get(taskId) {
      return handleReadSingle("cop_tasks", taskId, tasksCache, 'tasks.get', 'task');
    },
    async list(params = {}) {
        // ... (implémentation list V2 omise pour concision)
        const { status, task_type, worker_agent_name, limit = 100 } = params;
        const fetchFromDatabase = async () => {
            let q = client.from("cop_tasks").select("*").order("created_at", { ascending: false }).limit(limit);
            if (status) q = q.eq("status", status);
            if (task_type) q = q.eq("task_type", task_type);
            if (worker_agent_name) q = q.eq("worker_agent_name", worker_agent_name);
            return q;
        };
        const cacheKey = `list-tasks-${status || 'all'}-${task_type || 'all'}-${worker_agent_name || 'all'}-${limit}`;
        if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
            const dbResult = await executeStorageOperation('Read', 'tasks.list', fetchFromDatabase);
            return { ...dbResult, tasks: dbResult.data || [] };
        }
        if (tasksCache.has(cacheKey)) {
            logOperation('Read', 'tasks.list', { ...params, cached: true });
            return { ok: true, tasks: tasksCache.get(cacheKey) };
        }
        const result = await executeStorageOperation('Read', 'tasks.list', fetchFromDatabase);
        if (result.ok) tasksCache.set(cacheKey, result.data);
        return { ...result, tasks: result.data || [] };
    },
  };

  const steps = {

      return handleUpsert("cop_steps", stepRecord, stepsCache, 'steps.upsert', 'id', 'id');
    },
    async update(taskId, stepId, patch) {
        // ... (implémentation update V2 omise pour concision)
        if (currentSyncMode !== 'all' && currentSyncMode !== 'writes' && stepsCache.has(taskId)) {
            const cachedSteps = stepsCache.get(taskId);
            const existingStep = cachedSteps.find(step => step.id === stepId);

            if (existingStep) {
                const potentialNewStep = { ...existingStep, ...patch };
                if (areObjectsEqual(potentialNewStep, existingStep)) {
                    logOperation('Write', 'steps.update', { stepId, skipped: true });
                    return { ok: true, step: existingStep };
                }
            }
        }

        const dbResult = await executeStorageOperation('Write', 'steps.update', () =>
            client.from("cop_steps").update(patch).eq("id", stepId).select().maybeSingle()
        );

        if (dbResult.ok && dbResult.data) {
            stepsCache.delete(dbResult.data.task_id);
        }

        return { ...dbResult, step: dbResult.data };
    },
    async listByTask(taskId) {

        const fetchFromDatabase = async () => {
            return client.from("cop_steps").select("*").eq("task_id", taskId).order("created_at", { ascending: true });
        };
        if (currentSyncMode === 'all' || currentSyncMode === 'reads') {
            const dbResult = await executeStorageOperation('Read', 'steps.listByTask', fetchFromDatabase);
            return { ...dbResult, steps: dbResult.data || [] };
        }
        if (stepsCache.has(taskId)) {
            logOperation('Read', 'steps.listByTask', { taskId, cached: true });
            return { ok: true, steps: stepsCache.get(taskId) };
        }
        const result = await executeStorageOperation('Read', 'steps.listByTask', fetchFromDatabase);
        if (result.ok) stepsCache.set(taskId, result.data);
        return { ...result, steps: result.data || [] };
    },
  };

  const storage = {
    options: { ...options, type: 'supabase' },
    debugLogs,
    events,
    artifacts,
    agentIdentities,
    tasks,
    steps,
    getCacheContents: () => ({
      agentIdentities: Array.from(agentIdentitiesCache.entries()),
      tasks: Array.from(tasksCache.entries()),
      steps: Array.from(stepsCache.entries()),
    }),
    clearCache: () => {
      logOperation('Cache', 'clearCache', { message: 'Manually clearing all caches' });
      flushAllCaches();
    },
    ERROR_CODES,
  };

  options.storage = storage;
  defaultStorage = storage;
  return storage;
}

export function getStorage(options) {
  const providedStorage = options?.storage;
  // Ajout du type 'redis' possible
  const storageType = options?.type?.toLowerCase() || 'supabase';

  const configMatches = (storageInstance) => {
    if (!options) return true;
    if (!storageInstance || !storageInstance.options) return false;

    // 1. La vérification principale est le type de stockage
    if (storageInstance.options.type !== storageType) {
        return false;
    }

    // 2. Vérification secondaire pour les options spécifiques à Supabase/Redis
    if (storageType === 'supabase') {
        let matches = true;
        if (options.supabaseUrl !== undefined && options.supabaseUrl !== storageInstance.options.supabaseUrl) {
            matches = false;
        }
        if (options.supabaseServiceKey !== undefined && options.supabaseServiceKey !== storageInstance.options.supabaseServiceKey) {
            matches = false;
        }
        return matches;
    }

    // 2b. Vérification pour Redis
    if (storageType === 'redis') {
        let matches = true;
        if (options.redisUrl !== undefined && options.redisUrl !== storageInstance.options.redisUrl) {
             matches = false;
        }
        return matches;
    }

    // 3. Pour les autres types, le match de type est suffisant
    return true;
  };

  // 1. Priorité : Instance explicitement fournie et qui correspond
  if (providedStorage && configMatches(providedStorage)) {
    if (providedStorage !== defaultStorage) {
      defaultStorage = providedStorage;
    }
    return providedStorage;
  }

  // 2. Priorité : Instance par défaut existante et qui correspond
  if (defaultStorage && configMatches(defaultStorage)) {
    return defaultStorage;
  }

  // 3. Initialiser une nouvelle instance en fonction du type
  let newStorage;
  switch (storageType) {
    case 'memory':
        newStorage = createInMemoryStorage();
        break;
    case 'indexeddb':
        if (typeof window !== 'undefined' || typeof self !== 'undefined') {
            newStorage = createIndexedDBStorage(options);
        } else {
            console.error("IndexedDB storage requested but running in a non-browser environment. Falling back to in-memory.");
            newStorage = createInMemoryStorage();
        }
        break;
    case 'sqlite':
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') {
            newStorage = createSqliteStorage(options);
        } else {
            console.error("SQLite storage requested but running in a browser environment. Falling back to in-memory.");
            newStorage = createInMemoryStorage();
        }
        break;
    case 'redis':
        // Redis est un cas Node/Deno/Server side
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') {
            newStorage = createRedisStorage(options);
        } else {
            console.error("Redis storage requested but running in a browser environment. Falling back to in-memory.");
            newStorage = createInMemoryStorage();
        }
        break;
    case 'supabase':
    default:
        newStorage = initStorage(options);
        break;
  }

  // L'objet factory doit définir le type dans ses options
  newStorage.options.type = storageType;
  defaultStorage = newStorage;
  return newStorage;
}

  defaultStorage = storage;

🎯 Le Stockage d'Objets Supabase (Storage Buckets)
Les Buckets de Supabase sont construits sur PostgreSQL + S3 (ou son équivalent compatible MinIO/Cloud Storage). Ils sont parfaitement conçus pour :

Séparer les données : Stocker les gros fichiers à l'extérieur de votre base de données relationnelle (cop_tasks, cop_steps), ce qui garantit que vos requêtes SQL restent rapides.

Scalabilité et Coût : Offrir un stockage optimisé pour les objets (fichiers), généralement moins cher et plus performant pour les I/O de gros fichiers que de les stocker en tant que bytea dans la base de données.

Sécurité : Permettre un contrôle d'accès fin via les politiques RLS (Row Level Security) de Supabase, ce qui est crucial pour un protocole d'orchestration.

JavaScript
// ... dans la fonction initStorage ...
  const fileStorage = {
    // Le bucket par défaut pour les artefacts du protocole COP
    defaultBucket: 'cop-artifacts',

    /**
     * @param {string} bucketName - Nom du bucket cible.
     * @param {string} path - Le chemin où stocker le fichier (ex: 'task_123/output.json').
     * @param {Buffer|File} fileBody - Le contenu du fichier.
     * @param {object} options - Options d'upload (ex: { contentType: 'application/json' })
     */
    async uploadArtifact(bucketName, path, fileBody, options = {}) {
        const bucket = bucketName || this.defaultBucket;
        return executeStorageOperation('Write', 'fileStorage.uploadArtifact', async () => {
            // NOTE: Si le bucket n'existe pas, il faudrait d'abord le créer ici ou lors de l'initialisation.
            const { data, error } = await client.storage
                .from(bucket)
                .upload(path, fileBody, options);

            if (error) throw error; // L'erreur sera gérée par executeStorageOperation

            return { path: data.path }; // Retourne le chemin complet
        }, { bucket, path });
    },

    async downloadArtifact(bucketName, path) {
        const bucket = bucketName || this.defaultBucket;
        return executeStorageOperation('Read', 'fileStorage.downloadArtifact', async () => {
            const { data, error } = await client.storage
                .from(bucket)
                .download(path);

            if (error) throw error;

            // data est un Blob, on le retourne pour l'environnement Node/Browser
            return { data: data };
        }, { bucket, path });
    },

    async getPublicUrl(bucketName, path) {
        const bucket = bucketName || this.defaultBucket;
        // Pas besoin de passer par executeStorageOperation si l'API est locale et ne fait pas d'E/S
        const { data } = client.storage.from(bucket).getPublicUrl(path);

        if (!data || !data.publicUrl) {
            return { ok: false, error: 'Failed to get public URL.', code: ERROR_CODES.DB_ERROR };
        }
        return { ok: true, url: data.publicUrl };
    }
  };

    events: { /* ... insert ... */ async insert(eventRecord) { inMemoryData.events.push(eventRecord); return { ok: true, event: eventRecord }; } },

export function getStorage(options) {
  const providedStorage = options?.storage;
  const storageType = options?.type?.toLowerCase() || 'supabase';

  const configMatches = (storageInstance) => { /* ... Logique de correspondance ... */ return true; };

  // 1. Priorité : Instance explicitement fournie et qui correspond
  if (providedStorage && configMatches(providedStorage)) {
    if (providedStorage !== defaultStorage) { defaultStorage = providedStorage; }
    return providedStorage;
  }

  // 2. Priorité : Instance par défaut existante et qui correspond
  if (defaultStorage && configMatches(defaultStorage)) {
    return defaultStorage;
  }

  // 3. Initialiser une nouvelle instance en fonction du type
  let newStorage;
  switch (storageType) {
    case 'memory':
        newStorage = createInMemoryStorage();
        break;
    case 'indexeddb':
        // Vérification de l'environnement (Navigateur)
        if (typeof window !== 'undefined' || typeof self !== 'undefined') { newStorage = createIndexedDBStorage(options); }
        else { console.error("IndexedDB requested in non-browser env. Falling back to in-memory."); newStorage = createInMemoryStorage(); }
        break;
    case 'sqlite':
        // Vérification de l'environnement (Node/Deno)
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') { newStorage = createSqliteStorage(options); }
        else { console.error("SQLite requested in browser env. Falling back to in-memory."); newStorage = createInMemoryStorage(); }
        break;
    case 'redis':
        // Vérification de l'environnement (Node/Deno/Server)
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') { newStorage = createRedisStorage(options); }
        else { console.error("Redis requested in browser env. Falling back to in-memory."); newStorage = createInMemoryStorage(); }
        break;
    case 'supabase':
    default:
        newStorage = initStorage(options);
        break;
  }

  newStorage.options.type = storageType;
  defaultStorage = newStorage;
  return newStorage;
}

  defaultStorage = storage;

export function getStorage(options) {
  const providedStorage = options?.storage;
  const storageType = options?.type?.toLowerCase() || 'supabase';

  const configMatches = (storageInstance) => {
    if (!options) return true;
    if (!storageInstance || !storageInstance.options) return false;
    if (storageInstance.options.type !== storageType) { return false; }

    if (storageType === 'supabase') {
        let matches = true;
        if (options.supabaseUrl !== undefined && options.supabaseUrl !== storageInstance.options.supabaseUrl) { matches = false; }
        if (options.supabaseServiceKey !== undefined && options.supabaseServiceKey !== storageInstance.options.supabaseServiceKey) { matches = false; }
        return matches;
    }

    if (storageType === 'redis') {
        let matches = true;
        if (options.redisUrl !== undefined && options.redisUrl !== storageInstance.options.redisUrl) { matches = false; }
        return matches;
    }

    // Les autres types (memory, sqlite, indexeddb) n'ont pas de vérification de config complexe
    return true;
  };

  // Logique de Singleton/Factory
  if (providedStorage && configMatches(providedStorage)) {
    if (providedStorage !== defaultStorage) { defaultStorage = providedStorage; }
    return providedStorage;
  }
  if (defaultStorage && configMatches(defaultStorage)) {
    return defaultStorage;
  }

  // Initialisation du nouveau stockage en fonction du type
  let newStorage;
  switch (storageType) {
    case 'memory':
        newStorage = createInMemoryStorage();
        break;
    case 'indexeddb':
        // Vérification de l'environnement (Navigateur)
        if (typeof window !== 'undefined' || typeof self !== 'undefined') { newStorage = createIndexedDBStorage(options); }
        else { console.error("IndexedDB requested in non-browser env. Falling back to in-memory."); newStorage = createInMemoryStorage(); }
        break;
    case 'sqlite':
        // Vérification de l'environnement (Node/Deno)
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') { newStorage = createSqliteStorage(options); }
        else { console.error("SQLite requested in browser env. Falling back to in-memory."); newStorage = createInMemoryStorage(); }
        break;
    case 'redis':
        // Vérification de l'environnement (Node/Deno/Server)
        if (typeof process !== 'undefined' || typeof Deno !== 'undefined') { newStorage = createRedisStorage(options); }
        else { console.error("Redis requested in browser env. Falling back to in-memory."); newStorage = createInMemoryStorage(); }
        break;
    case 'supabase':
    default:
        newStorage = initStorage(options);
        break;
  }

  newStorage.options.type = storageType;
  defaultStorage = newStorage;
  return newStorage;
}

  defaultStorage = storage;

2. Le Cœur (Logique Métier & Audit)
Fichier : lib/storage/StorageCore.js

C'est ici que réside la valeur ajoutée "juridique". Ce module enveloppe la donnée brute avec des métadonnées inviolables (Timestamp, Hash SHA-256) et assure le chaînage (le hash de l'entrée N dépend du hash de l'entrée N-1), rendant toute modification de l'historique mathématiquement détectable.

JavaScript
const crypto = require('crypto');

class StorageCore {
    /**
     * @param {Object} adapter - Une instance de StorageAdapterInterface.
     */
    constructor(adapter) {
        this.adapter = adapter;
    }

    async init() {
        await this.adapter.init();
    }

    /**
     * Calcule le Hash SHA-256 d'une donnée.
     * @private
     */
    _generateHash(data) {
        return crypto.createHash('sha256').update(JSON.stringify(data)).digest('hex');
    }

    /**
     * Sauvegarde un état ou un événement pour un acteur donné.
     * Implémente le "Ledgering" : chaque entrée est liée à la précédente.
     * * @param {string} actorId
     * @param {Object} payload - Les données métier.
     * @param {string} [eventType="STATE_CHANGE"] - Type d'événement (pour filtrage futur).
     */
    async persist(actorId, payload, eventType = "STATE_CHANGE") {
        // 1. Récupérer le dernier hash connu pour cet acteur (Chain integrity)
        const history = await this.adapter.retrieveHistory(actorId);
        const lastEntry = history.length > 0 ? history[history.length - 1] : null;
        const previousHash = lastEntry ? lastEntry.integrityHash : "GENESIS_0000000000000000000000000";

        // 2. Création de l'enveloppe d'audit
        const timestamp = new Date().toISOString();

        const auditData = {
            actorId,
            eventType,
            payload,
            meta: {
                timestamp,
                previousHash,
                version: "1.0"
            }
        };

        // 3. Scellement cryptographique (Hash de l'enveloppe entière)
        const integrityHash = this._generateHash(auditData);

        // 4. Construction de l'enregistrement final immuable
        const finalRecord = {
            ...auditData,
            integrityHash // Ce hash scelle le contenu ET le lien vers le passé
        };

        // 5. Délégation au stockage physique
        await this.adapter.append(actorId, finalRecord);

        return finalRecord;
    }

    /**
     * Récupère l'état courant (reconstruction depuis l'historique)
     * Dans un modèle Event Sourcing pur, on replay tout.
     * Ici, on retourne l'historique complet pour audit.
     */
    async getAuditLog(actorId) {
        return await this.adapter.retrieveHistory(actorId);
    }

    /**
     * Vérifie l'intégrité cryptographique de la chaîne d'un acteur.
     * Utile pour un audit juridique.
     */
    async verifyIntegrity(actorId) {
        const history = await this.adapter.retrieveHistory(actorId);
        let previousHash = "GENESIS_0000000000000000000000000";

        for (const [index, record] of history.entries()) {
            // 1. Vérifier le lien de chaîne
            if (record.meta.previousHash !== previousHash) {
                throw new Error(`Integrity Broken at index ${index}: Previous hash mismatch.`);
            }

            // 2. Recalculer le hash du contenu
            const { integrityHash, ...dataToHash } = record;
            const calculatedHash = this._generateHash(dataToHash);

            if (calculatedHash !== integrityHash) {
                throw new Error(`Integrity Broken at index ${index}: Content modified.`);
            }

            previousHash = integrityHash;
        }
        return true; // Chaîne valide
    }

    async shutdown() {
        await this.adapter.close();
    }
}

module.exports = StorageCore;
3. L'Adaptateur Système de Fichiers (Append-Only)
Fichier : lib/storage/adapters/FileSystemAdapter.js

Implémentation d'un stockage fichier local. Chaque acteur possède son fichier .jsonl (JSON Lines). C'est un format idéal pour les logs : robuste (on append juste à la fin), lisible par un humain, et facile à parser ligne par ligne (stream).

JavaScript
const fs = require('fs/promises');
const path = require('path');
const StorageAdapterInterface = require('../StorageInterface');

class FileSystemAdapter extends StorageAdapterInterface {
    /**
     * @param {Object} options
     * @param {string} options.baseDir - Dossier racine du stockage.
     */
    constructor(options = {}) {
        super();
        this.baseDir = options.baseDir || path.join(process.cwd(), 'data', 'store');
    }

    async init() {
        // Assure que le dossier de stockage existe
        try {
            await fs.mkdir(this.baseDir, { recursive: true });
        } catch (err) {
            if (err.code !== 'EEXIST') throw err;
        }
    }

    _getFilePath(actorId) {
        // Sanitize actorId pour éviter les traversées de dossiers
        const safeId = actorId.replace(/[^a-z0-9-_]/gi, '_');
        return path.join(this.baseDir, `${safeId}.jsonl`);
    }

    async append(actorId, record) {
        const filePath = this._getFilePath(actorId);
        const line = JSON.stringify(record) + '\n';

        // flag 'a' pour append. Atomicité garantie par l'OS pour les écritures < PIPE_BUF (généralement 4KB)
        // Pour une robustesse absolue, on pourrait implémenter un file-lock,
        // mais le modèle acteur tend à sérialiser les écritures par acteur naturellement.
        await fs.appendFile(filePath, line, { encoding: 'utf8', flag: 'a' });
    }

    async retrieveHistory(actorId) {
        const filePath = this._getFilePath(actorId);
        try {
            const data = await fs.readFile(filePath, 'utf8');
            // Trim pour éviter la ligne vide à la fin, split par newline, parse JSON
            return data.trim().split('\n').map(line => JSON.parse(line));
        } catch (err) {
            if (err.code === 'ENOENT') {
                return []; // Pas encore d'historique
            }
            throw err;
        }
    }

    async close() {
        // Rien de spécifique à fermer pour fs/promises, mais utile pour d'autres DBs
        return Promise.resolve();
    }
}

module.exports = FileSystemAdapter;

Analyse technique pour le développeur expert
Immuabilité et Auditabilité (Opposabilité) :

Dans StorageCore, j'ai implémenté un système de Blockchain-lite. L'inclusion de previousHash dans le calcul du integrityHash courant garantit que si une ligne du fichier .jsonl est supprimée ou modifiée manuellement, toute la chaîne suivante devient mathématiquement invalide lors du verifyIntegrity().

Format JSONL :

Le choix du JSON Lines (.jsonl) dans FileSystemAdapter est délibéré. Contrairement à un gros fichier JSON qui nécessite de lire/parser/réécrire tout le fichier à chaque ajout (O(n)), le JSONL permet un append atomique O(1). C'est crucial pour la performance des acteurs.
